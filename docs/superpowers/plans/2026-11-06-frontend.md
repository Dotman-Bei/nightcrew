async function floorRuns() {
  const { chromium } = await import("playwright-core");
  const candidates = [process.env.CHROME_PATH, "C:/Program Files/Google/Chrome/Application/chrome.exe", "/usr/bin/google-chrome"];
  const browser = await chromium.launch({ executablePath: candidates.find((path) => path && existsSync(path)), headless: true });
  try {
    const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });
    await page.goto(url, { waitUntil: "load" });
    await page.waitForTimeout(1500);
    const sandbox = await page.locator("iframe").first().getAttribute("sandbox");
    const frame = page.frames().find((candidate) => candidate.url().includes("/f/"));
    const text = frame ? (await frame.locator("body").textContent({ timeout: 5000 }))?.trim() : "";
    const origin = frame ? await frame.evaluate(() => self.origin) : "";
    console.log(`floor frame: sandbox="${sandbox}", origin=${origin}, ${text?.length ?? 0} characters of text`);
    if (sandbox !== "allow-scripts") throw new Error("the frame must be sandboxed with allow-scripts only");
    if (!frame || !text) throw new Error("the floor did not render in its frame");
    if (origin !== "null") throw new Error("the floor must have an opaque origin");
    return 0;
  } catch (error) {
    console.error(error instanceof Error ? error.message : error);
    return 1;
  } finally {
    await browser.close();
  }
}

const KIT_SCRIPTS = ["/taste/system.js", "/taste/motion/motion.js", "/taste/layout/layout.js", "/taste/diagrams/mount.js"];

/**
 * The kit's scripts change the page's DOM (effect canvases, pause buttons, nav sheets), so they load
 * after React has hydrated. Loaded earlier, they race hydration and cause mismatch errors.

export function KitScripts() {
  useEffect(() => {
    for (const src of KIT_SCRIPTS) {
      const script = document.createElement("script");
      script.type = "module";

      document.body.appendChild(script);

  }, []);

import { useState } from "react";

/**
 * The floor in its sandbox. CSP cannot stop a frame navigating itself, so a second load means the
 * floor tried to leave: the frame is removed. No allow-same-origin, ever.
 */
export function FloorFrame({ src, title }: { src: string; title: string }) {
  const [loads, setLoads] = useState(0);
  if (loads > 1) {
    return (
      <div className="floor-frame__left">
        <p>This floor tried to leave its sandbox, so it was removed from the page.</p>
      </div>
    );
  }
  return (
    <div className="stage">
      <iframe className="floor-frame" title={title} src={src} sandbox="allow-scripts" allow="" referrerPolicy="no-referrer" onLoad={() => setLoads((count) => count + 1)} />
    </div>
  );
}

function short(address: string): string {

/** The one filled button on the page until a wallet is connected. */
export function ConnectPanel() {
  const { address, isConnected } = useAccount();

  const { disconnect } = useDisconnect();
  if (isConnected && address) {

    </div>

async function send(path: string, body: unknown): Promise<{ ok: boolean; error?: string; existingId?: number }> {
  const response = await fetch(path, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) });
  const parsed = (await response.json().catch(() => ({}))) as { error?: string; existingId?: number };
  return { ok: response.ok, ...parsed };
}

export function VoteButton({ proposalId, shiftSlug }: { proposalId: number; shiftSlug: string }) {
  const { address, isConnected } = useAccount();
  const { signMessageAsync } = useSignMessage();
  const router = useRouter();
  const [state, setState] = useState<"idle" | "signing" | "done">("idle");
  const [problem, setProblem] = useState<string | null>(null);

  async function vote() {
    if (!address) return;
    setProblem(null);
    setState("signing");
    try {
      const signedAt = new Date().toISOString();
      const signature = await signMessageAsync({ message: voteMessage({ proposalId, shiftSlug, signedAt }) });
      const result = await send("/api/votes", { proposalId, shiftSlug, address, signature, signedAt });
      if (!result.ok) throw new Error(result.error ?? "That vote did not go through.");
      setState("done");
      router.refresh();
    } catch (error) {
      setProblem(error instanceof Error ? error.message : "That vote did not go through.");
      setState("idle");
    }
  }

  return (
    <span className="vote-button">
      <button className="btn btn--secondary" type="button" disabled={!isConnected || state === "signing"} onClick={vote}>{state === "done" ? "Voted" : state === "signing" ? "Sign in your wallet" : "Vote"}</button>
      {problem ? <span role="alert" className="vote-button__problem">{problem}</span> : null}
    </span>
  );
}

export function ProposeForm() {
  const { address, isConnected } = useAccount();
  const { signMessageAsync } = useSignMessage();

    event.preventDefault();
    if (!address) return;

      setState("done");
      form.reset();
      router.refresh();
    } catch (error) {
      setProblem(error instanceof Error ? error.message : "That proposal did not go through.");
      setState("idle");
    }
  }

  return (
    <form className="form" onSubmit={submit}>
      <label className="field field--full">One sentence, up to 200 characters<textarea name="text" required maxLength={200} rows={3} /></label>

  );
}

"use client";

import { useState, type FormEvent } from "react";

export function WaitlistForm() {
  const [state, setState] = useState<"idle" | "sending" | "sent" | "error">("idle");
  const [message, setMessage] = useState("");

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const data = new FormData(event.currentTarget);
    setState("sending");
    const response = await fetch("/api/waitlist", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ community: data.get("community"), contact: data.get("contact") }),
    });
    if (response.ok) {
      setState("sent");
    } else {
      const body = (await response.json().catch(() => ({}))) as { error?: string };
      setMessage(body.error ?? "That did not go through. Try again.");
      setState("error");
    }
  }

  if (state === "sent") return <p className="t-lead" role="status">You are on the list. We will write when a second crew opens.</p>;
  return (
    <form className="form" onSubmit={submit}>
      <label className="field field--full">Community or token name<input name="community" required maxLength={80} autoComplete="organization" /></label>
      <label className="field field--full">How to reach you<input name="contact" required maxLength={120} placeholder="An email or a handle" /></label>
      <div className="btn-row">
        <button className="btn btn--secondary" type="submit" disabled={state === "sending"}>{state === "sending" ? "Sending" : "Join the waitlist"}</button>
      </div>
      {state === "error" ? <p role="alert">{message}</p> : null}
    </form>
  );
}

**The defect.** `FloorFrame` counts the frame's `load` events and removes the frame on a second one, because CSP and the sandbox cannot stop a frame navigating itself (spec section 6). The frame is in the server's HTML, so it starts loading before React attaches `onLoad`. Any load before hydration goes uncounted. Nothing is exposed, because the sandbox has no `allow-same-origin` and no top navigation.

A `noscript` line covers a browser without JavaScript, which would otherwise see an empty box. The direct URL `/f/<hash>` and the floor screenshots do not use this component and do not change.

"use client";

import { useEffect, useState } from "react";

/**
 * The floor in its sandbox. CSP cannot stop a frame navigating itself, so a second load means the
 * floor tried to leave: the frame is removed. No allow-same-origin, ever.
 *
 * The frame is created only after hydration. A frame in the server's HTML starts loading before
 * React can attach onLoad, so its first loads go uncounted and a floor that leaves stays on the page.
 */
export function FloorFrame({ src, title }: { src: string; title: string }) {
  const [hydrated, setHydrated] = useState(false);
  const [loads, setLoads] = useState(0);
  useEffect(() => setHydrated(true), []);
  if (loads > 1) {
    return (
      <div className="stage">
        <div className="floor-frame__left">
          <p>This floor tried to leave its sandbox, so it was removed from the page.</p>
        </div>
      </div>
    );
  }
  return (
    <div className="stage">
      {hydrated ? (
        <iframe className="floor-frame" title={title} src={src} sandbox="allow-scripts" allow="" referrerPolicy="no-referrer" onLoad={() => setLoads((count) => count + 1)} />
      ) : (
        <noscript>
          <div className="floor-frame__left">
            <p>This floor runs in your browser, so it needs JavaScript.</p>
          </div>
        </noscript>
      )}
    </div>
  );
}
