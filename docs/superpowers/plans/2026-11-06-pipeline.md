# Shift pipeline Implementation Plan

**Goal:** Run a whole shift unattended: close the vote, triage, build in a tool loop with no authority, review, verify on a clean machine, publish a receipt and attest it on Base.
**Architecture:** Four GitHub Actions jobs, each holding only what it needs. Close holds the database, Crew holds only the OpenRouter key, Verify holds nothing, and Publish holds the database and the attester key. Jobs pass files, and every file is validated as hostile data. Everything can be proven with a fake crew and a fake chain before the user's key exists.
**Tech Stack:** @anthropic-ai/sdk 1.6.0 pointed at OpenRouter, viem 2.57.16, zod 4.6.7, diff 9.0.0, playwright-core 1.65.0, GitHub Actions on ubuntu-latest
**Source spec:** docs/superpowers/specs/2026-11-06-nightcrew-design.md

Task references use `<plan>:<task>`, across all six plans, in build order: `foundation`, `floor`, `pipeline`, `votes`, `frontend`, `launch`. The code here passed each task's gate in a reference build on 2026-11-06. Tasks marked as cut candidates can be skipped at the Sat 7 Nov 12:00 UTC checkpoint, through the architect, never by hand. Gates need the installed dependencies: a fresh worktree runs `npm ci` first, with a real `node_modules` folder and never a link to another one.

**Amended 2026-11-06: the crew runs on OpenRouter, not Managed Agents.** Tasks already built keep their original text, so some still name Managed Agents: `foundation:7`, `pipeline:14` and `pipeline:15` amend them. Tasks not yet built were edited in place, and every task id is unchanged.

```ts
import { z } from "zod";

const WEI_PER_VIBES = 10n ** 18n;

export const StakeMode = z.enum(["current-at-tally", "unavailable", "omitted"]);
export type StakeMode = z.infer<typeof StakeMode>;

export type StakeReader = (addresses: string[]) => Promise<{ staked: Map<string, bigint>; mode: StakeMode }>;

/** Used if the staking half of vote weight is cut: every stake counts as zero. */
export const noStakeReader: StakeReader = async () => ({ staked: new Map(), mode: "omitted" });

export const TallyEntry = z.object({
  proposalId: z.number(),
  text: z.string(),
  author: z.string(),
  seeded: z.boolean(),
  weight: z.string(),
  wallets: z.number(),
});
export type TallyEntry = z.infer<typeof TallyEntry>;

export const VoterRecord = z.object({
  address: z.string(),
  proposalId: z.number(),
  held: z.string(),
  staked: z.string(),
  weight: z.string(),
});
export type VoterRecord = z.infer<typeof VoterRecord>;

export const Tally = z.object({
  snapshotBlock: z.string().nullable(),
  closedAt: z.string(),
  stakeMode: StakeMode,
  entries: z.array(TallyEntry),
  voters: z.array(VoterRecord),
});
export type Tally = z.infer<typeof Tally>;

/** max(1, whole $VIBES held plus staked). A wallet with none still counts at minimal weight. */
export function voteWeight(held: bigint, staked: bigint): bigint {
  const whole = (held + staked) / WEI_PER_VIBES;
  return whole < 1n ? 1n : whole;
}

export type TallyInput = {
  proposals: { id: number; text: string; author: string; seeded: boolean }[];
  votes: { voter: string; proposalId: number }[];
  held: Map<string, bigint>;
  staked: Map<string, bigint>;
  stakeMode: StakeMode;
  snapshotBlock: string | null;
  closedAt: string;
};

/** Ranks weight descending, then wallets descending, then oldest proposal first. */
export function computeTally(input: TallyInput): Tally {
  const open = new Map(input.proposals.map((proposal) => [proposal.id, proposal]));
  const totals = new Map<number, { weight: bigint; wallets: number }>();
  const voters: VoterRecord[] = [];
  for (const vote of input.votes) {
    if (!open.has(vote.proposalId)) continue;
    const held = input.held.get(vote.voter) ?? 0n;
    const staked = input.staked.get(vote.voter) ?? 0n;
    const weight = voteWeight(held, staked);
    const total = totals.get(vote.proposalId) ?? { weight: 0n, wallets: 0 };
    totals.set(vote.proposalId, { weight: total.weight + weight, wallets: total.wallets + 1 });
    voters.push({ address: vote.voter, proposalId: vote.proposalId, held: held.toString(), staked: staked.toString(), weight: weight.toString() });
  }
  const entries = input.proposals
    .map((proposal) => {
      const total = totals.get(proposal.id) ?? { weight: 0n, wallets: 0 };
      return { proposalId: proposal.id, text: proposal.text, author: proposal.author, seeded: proposal.seeded, weight: total.weight, wallets: total.wallets };
    })
    .sort((a, b) => {
      if (a.weight !== b.weight) return a.weight > b.weight ? -1 : 1;
      if (a.wallets !== b.wallets) return b.wallets - a.wallets;
      return a.proposalId - b.proposalId;
    })
    .map((entry) => ({ ...entry, weight: entry.weight.toString() }));
  return { snapshotBlock: input.snapshotBlock, closedAt: input.closedAt, stakeMode: input.stakeMode, entries, voters };
}
```

```ts
import { createPublicClient, erc20Abi, http, type Address } from "viem";
import { base } from "viem/chains";
import { VIBES_ADDRESS, baseRpcUrl } from "@/lib/config";

export type BlockInfo = { number: bigint; timestamp: bigint };

export type ChainReader = {
  latestBlock(): Promise<BlockInfo>;
  blockAt(number: bigint): Promise<BlockInfo>;
  /** $VIBES balance of each address at the block, in wei, keyed by lower-case address. */
  heldAt(addresses: string[], block: bigint): Promise<Map<string, bigint>>;
};

const BASE_BLOCK_SECONDS = 2n;

/** The last block whose timestamp is at or before `target`, by estimate then binary search. */
export async function findSnapshotBlock(reader: ChainReader, target: bigint): Promise<bigint> {
  const latest = await reader.latestBlock();
  if (latest.timestamp <= target) return latest.number;
  const estimate = latest.number - (latest.timestamp - target) / BASE_BLOCK_SECONDS;
  let width = 64n;
  let low = estimate - width < 0n ? 0n : estimate - width;
  let high = estimate + width > latest.number ? latest.number : estimate + width;
  while (low > 0n && (await reader.blockAt(low)).timestamp > target) {
    width *= 2n;
    low = low - width < 0n ? 0n : low - width;
  }
  while (high < latest.number && (await reader.blockAt(high)).timestamp <= target) {
    width *= 2n;
    high = high + width > latest.number ? latest.number : high + width;
  }
  while (low < high) {
    const middle = (low + high + 1n) / 2n;
    if ((await reader.blockAt(middle)).timestamp <= target) low = middle;
    else high = middle - 1n;
  }
  return low;
}

export function viemChainReader(rpcUrl: string = baseRpcUrl()): ChainReader {
  const client = createPublicClient({ chain: base, transport: http(rpcUrl) });
  return {
    async latestBlock() {
      const block = await client.getBlock();
      return { number: block.number, timestamp: block.timestamp };
    },
    async blockAt(number) {
      const block = await client.getBlock({ blockNumber: number });
      return { number: block.number, timestamp: block.timestamp };
    },
    async heldAt(addresses, block) {
      const held = new Map<string, bigint>();
      if (addresses.length === 0) return held;
      const results = await client.multicall({
        allowFailure: false,
        blockNumber: block,
        contracts: addresses.map((address) => ({
          address: VIBES_ADDRESS,
          abi: erc20Abi,
          functionName: "balanceOf" as const,
          args: [address as Address] as const,
        })),
      });
      addresses.forEach((address, index) => held.set(address.toLowerCase(), results[index]));
      return held;
    },
  };
}
```

```ts
import { VIBESTARTER_API } from "@/lib/config";
import type { StakeReader } from "@/lib/tally";

/** Current-only: the API has no history, so receipts label the figure "current at tally". */
export function apiStakeReader(fetchImpl: typeof fetch = fetch): StakeReader {
  return async (addresses) => {
    const staked = new Map<string, bigint>();
    try {
      for (const address of addresses) {
        const response = await fetchImpl(`${VIBESTARTER_API}/api/staking/stats?wallet=${address}`);
        if (!response.ok) throw new Error(`staking API returned ${response.status}`);
        const body = (await response.json()) as { stakedBalance?: string };
        staked.set(address.toLowerCase(), BigInt(body.stakedBalance ?? "0"));
      }
      return { staked, mode: "current-at-tally" };
    } catch {
      return { staked: new Map(), mode: "unavailable" };
    }
  };
}
```

```ts
import { VIBESTARTER_API } from "@/lib/config";

const MAX_BYTES = 100_000;
const MAX_CAMPAIGNS = 20;
const ZERO_WALLET = "0x0000000000000000000000000000000000000001";

function primitivesOnly(value: unknown): Record<string, string | number | boolean> {
  const out: Record<string, string | number | boolean> = {};
  if (!value || typeof value !== "object") return out;
  for (const [key, item] of Object.entries(value)) {
    if (typeof item === "number" || typeof item === "boolean") out[key] = item;
    if (typeof item === "string" && item.length <= 200 && !/https?:/i.test(item)) out[key] = item;
  }
  return out;
}

/** Public figures a floor may read, because a floor cannot call anything itself. Never throws. */
export async function fetchFloorData(fetchImpl: typeof fetch = fetch, now: Date = new Date()): Promise<string> {
  const data: Record<string, unknown> = { fetchedAt: now.toISOString(), source: "Public read endpoints, fetched for this shift", errors: [] };
  const errors = data.errors as string[];
  const get = async (path: string): Promise<unknown> => {
    try {
      const response = await fetchImpl(`${VIBESTARTER_API}${path}`);
      if (!response.ok) throw new Error(`status ${response.status}`);
      return await response.json();
    } catch (error) {
      errors.push(`${path}: ${error instanceof Error ? error.message : "failed"}`);
      return null;
    }
  };
  data.stats = primitivesOnly(await get("/api/stats"));
  const ledger = (await get("/api/ledger/stats")) as { cells?: unknown } | null;
  data.ledger = primitivesOnly(ledger?.cells);
  data.circulatingSupply = await get("/api/supply/circulating");
  const staking = (await get(`/api/staking/stats?wallet=${ZERO_WALLET}`)) as { totalStaked?: string } | null;
  data.totalStaked = staking?.totalStaked ?? null;
  data.totalSupply = "1000000000000000000000000000";
  const campaigns = (await get("/api/campaigns")) as { campaigns?: unknown[] } | null;
  data.campaigns = (campaigns?.campaigns ?? []).slice(0, MAX_CAMPAIGNS).map(primitivesOnly);
  const text = JSON.stringify(data);
  if (text.length <= MAX_BYTES) return text;
  data.campaigns = [];
  errors.push("campaigns dropped: the snapshot was too large");
  return JSON.stringify(data);
}
```

```ts
import type { ChainReader } from "@/lib/chain";
```

```ts
import type { Db } from "@/lib/db";
import { claimOffSchedule, claimScheduled, createRehearsal, ensureOpenShift, spentCents } from "@/lib/shifts";
import { computeTally, type StakeReader, type Tally } from "@/lib/tally";
import type { Shift } from "@/lib/types";
import type { ShiftInput } from "@/crew/lib/contracts";

export type Trigger =
  | { kind: "scheduled" }
  | { kind: "off-schedule"; reason: string }
  | { kind: "rehearsal"; reason: string; proposal: string; author: string };

export type CloseDeps = {
  db: Db;
  chain: ChainReader;
  stakes: StakeReader;
  floorData: () => Promise<string>;
  now: Date;
  trigger: Trigger;
  capCents: number;
  totalBudgetCents: number;
};

type OpenProposal = { id: number; text: string; author: string; seeded: boolean };

async function publicTally(db: Db, chain: ChainReader, stakes: StakeReader, shift: Shift): Promise<Tally> {
  const proposals = await db.query<OpenProposal>(
    "select id::int as id, text, author, seeded from proposals where status = 'open' and failed_attempts < 2 order by id",
  );
  const votes = await db.query<{ voter: string; proposalId: number }>(
    'select voter, proposal_id::int as "proposalId" from votes where shift_id = $1',
    [shift.id],
  );
  const closesAt = new Date(shift.closes_at);
  const snapshotBlock = await findSnapshotBlock(chain, BigInt(Math.floor(closesAt.getTime() / 1000)));
  const voters = votes.rows.map((vote) => vote.voter);
  const held = await chain.heldAt(voters, snapshotBlock);
  const { staked, mode } = await stakes(voters);
  return computeTally({
    proposals: proposals.rows,
    votes: votes.rows,
    held,
    staked,
    stakeMode: mode,
    snapshotBlock: snapshotBlock.toString(),
    closedAt: closesAt.toISOString(),
  });
}

/** The close job. Returns null when there was nothing to claim. */
export async function closeShift(deps: CloseDeps): Promise<ShiftInput | null> {
  const { db, chain, stakes, floorData, now, trigger, capCents, totalBudgetCents } = deps;
  let shift: Shift | null;
  let rehearsalProposal: OpenProposal | null = null;
  if (trigger.kind === "scheduled") {
    shift = await claimScheduled(db, now);
  } else if (trigger.kind === "off-schedule") {
    shift = await claimOffSchedule(db, now, trigger.reason);
  } else {
    const created = await createRehearsal(db, now, trigger.reason, trigger.proposal, trigger.author);
    shift = created.shift;
    rehearsalProposal = { id: created.proposalId, text: trigger.proposal, author: trigger.author.toLowerCase(), seeded: true };
  }
  if (!shift) return null;
  await ensureOpenShift(db, now);

  const base = {
    shift: { id: shift.id, slug: shift.slug, label: shift.label, closesAt: new Date(shift.closes_at).toISOString(), startedAt: now.toISOString() },
  };
  try {
    const snapshot = await floorData();
    const tally = rehearsalProposal
      ? computeTally({ proposals: [rehearsalProposal], votes: [], held: new Map(), staked: new Map(), stakeMode: "omitted", snapshotBlock: null, closedAt: now.toISOString() })
      : await publicTally(db, chain, stakes, shift);
    const spent = await spentCents(db);
    let darkReason: string | null = null;
    if (spent + capCents > totalBudgetCents) darkReason = "credit budget reached";
    else if (tally.entries.length === 0) darkReason = "no proposals";
    return { ...base, tally, snapshot, darkReason, failure: null };
  } catch (error) {
    const reason = error instanceof Error ? error.message.slice(0, 400) : "the close job failed";
    return { ...base, tally: null, snapshot: "{}", darkReason: null, failure: { stage: "close", reason } };
  }
}
```

```ts
/** Dollars per 1M tokens, read from Anthropic's pricing page on 2026-11-06. */
const PRICES: Record<string, { input: number; cacheWrite: number; cacheRead: number; output: number }> = {
  "claude-opus-5-5": { input: 4, cacheWrite: 5, cacheRead: 0.2, output: 20 },
  "claude-sonnet-5-5": { input: 2, cacheWrite: 2.5, cacheRead: 0.1, output: 10 },
  "claude-haiku-5-5": { input: 0.1, cacheWrite: 0.125, cacheRead: 0.01, output: 0.5 },
};

export type TokenUsage = {
  input_tokens: number;
  output_tokens: number;
  cache_creation_input_tokens?: number | null;
  cache_read_input_tokens?: number | null;
};

/** The cost of one Messages API call in cents, as a fraction. Sum these and round up once. */
export function usageCents(model: string, usage: TokenUsage): number {
  const price = PRICES[model];
  if (!price) throw new Error(`no price for ${model}`);
  const dollars =
    (usage.input_tokens * price.input +
      (usage.cache_creation_input_tokens ?? 0) * price.cacheWrite +
      (usage.cache_read_input_tokens ?? 0) * price.cacheRead +
      usage.output_tokens * price.output) /
    1_000_000;
  return dollars * 100;
}
```

```ts
export type JsonSchema = Record<string, unknown>;

export type StructuredCall = {
  model: string;
  system: string;
  user: string;
  schema: JsonSchema;
  maxTokens: number;
};

export type StructuredResult = {
  /** Null when the model refused or returned nothing parseable. */
  value: unknown;
  refused: boolean;
  costCents: number;
};

/** A single call with no tools, whose reply is constrained to a schema. */
export type StructuredModel = (call: StructuredCall) => Promise<StructuredResult>;

export function anthropicStructuredModel(client: Anthropic = new Anthropic()): StructuredModel {
  return async ({ model, system, user, schema, maxTokens }) => {
    const message = await client.messages.create({
      model,
      max_tokens: maxTokens,
      system,
      messages: [{ role: "user", content: user }],
      output_config: { effort: "low", format: { type: "json_schema", schema } },
    });
    const costCents = usageCents(model, message.usage);
    if (message.stop_reason === "refusal") return { value: null, refused: true, costCents };
    const text = message.content.flatMap((block) => (block.type === "text" ? [block.text] : [])).join("");
    try {
      return { value: JSON.parse(text), refused: false, costCents };
    } catch {
      return { value: null, refused: false, costCents };
    }
  };
}
```

```ts
import type { TranscriptEntry } from "@/crew/lib/contracts";

const MAX_ENTRY_CHARS = 20_000;
const KEY_PATTERNS = [/sk-ant-[A-Za-z0-9_-]{8,}/g, /\b0x[a-fA-F0-9]{64}\b/g];

export type RawEvent = { type: string; [key: string]: unknown };

function blocksToText(content: unknown): string {
  if (typeof content === "string") return content;
  if (!Array.isArray(content)) return "";
  return content.map((block: { type?: string; text?: string }) => (block?.type === "text" ? (block.text ?? "") : `[${block?.type ?? "block"}]`)).join("\n");
}

function entry(event: RawEvent, kind: TranscriptEntry["kind"], text: string, tool: string | null = null, isError = false): TranscriptEntry {
  const truncated = text.length > MAX_ENTRY_CHARS;
  return {
    at: String(event.processed_at ?? event.created_at ?? ""),
    kind,
    tool,
    text: truncated ? `${text.slice(0, MAX_ENTRY_CHARS)}\n[truncated]` : text,
    truncated,
    isError,
  };
}

/** Turns session events into the public transcript. Thinking has no content, so it is left out. */
export function normaliseEvents(events: RawEvent[]): TranscriptEntry[] {
  const out: TranscriptEntry[] = [];
  for (const event of events) {
    switch (event.type) {
      case "user.message":
        out.push(entry(event, "prompt", blocksToText(event.content)));
        break;
      case "agent.message":
        out.push(entry(event, "message", blocksToText(event.content)));
        break;
      case "agent.tool_use":
        out.push(entry(event, "tool_use", JSON.stringify(event.input ?? {}), String(event.name ?? "tool")));
        break;
      case "agent.tool_result":
        out.push(entry(event, "tool_result", blocksToText(event.content), null, Boolean(event.is_error)));
        break;
      case "session.error":
        out.push(entry(event, "error", JSON.stringify(event.error ?? event)));
        break;
      case "session.status_idle":
        out.push(entry(event, "status", `idle: ${String((event.stop_reason as { type?: string } | undefined)?.type ?? "unknown")}`));
        break;
      default:
        break;
    }
  }
  return out;
}

/** Removes anything key-shaped. The sandbox holds no secrets, so a match is a defect worth seeing. */
export function scrubTranscript(entries: TranscriptEntry[]): { entries: TranscriptEntry[]; removed: number } {
  let removed = 0;
  const clean = entries.map((item) => {
    let text = item.text;
    for (const pattern of KEY_PATTERNS) {
      text = text.replace(pattern, () => {
        removed += 1;
        return "[removed]";
      });
    }
    return { ...item, text };
  });
  if (removed === 0) return { entries: clean, removed };
  const marker: TranscriptEntry = {
    at: clean[clean.length - 1]?.at ?? "",
    kind: "status",
    tool: null,
    text: `Removed ${removed} key-shaped string${removed === 1 ? "" : "s"} from this transcript. The sandbox holds no secrets, so this is a defect worth seeing.`,
    truncated: false,
    isError: true,
  };
  return { entries: [...clean, marker], removed };
}
```

```ts
export type CrewOutput = {
  result: CrewResult;
  transcript: TranscriptEntry[];
  html: string | null;
  floorJson: string | null;
};

export type CrewDeps = {
  input: ShiftInput;
  model: StructuredModel;
  runtime: CrewRuntime;
  kitFiles: KitFile[];
  buildModel: string;
};

/** Triage, build, review. Never throws: every failure becomes a published result. */
export async function crewShift({ input, model, runtime, kitFiles, buildModel }: CrewDeps): Promise<CrewOutput> {
  const triage: TriageDecision[] = [];
  const reviews: ReviewDecision[] = [];
  let cost = 0;
  let sessionId: string | null = null;
  let transcript: TranscriptEntry[] = [];
  let proposalId: number | null = null;
  const models = { triage: CREW_SUPPORT_MODEL, build: null as string | null, review: null as string | null };

  const finish = (partial: Pick<CrewResult, "outcome" | "failure" | "darkReason">, html: string | null = null, floorJson: string | null = null): CrewOutput => ({
    result: { ...partial, proposalId, triage, reviews, models, sessionId, costCents: Math.ceil(cost) },
    transcript,
    html,
    floorJson,
  });
  const fail = (stage: "triage" | "build" | "review", reason: string): CrewOutput =>
    finish({ outcome: "failed", failure: { stage, reason: reason.slice(0, 500) }, darkReason: null });

  let accepted: { id: number; text: string; brief: string } | null = null;
  try {
    for (const entry of (input.tally?.entries ?? []).slice(0, MAX_TRIAGE)) {
      const { decision, costCents } = await triageProposal(model, entry.proposalId, entry.text);
      cost += costCents;
      triage.push(decision);
      if (decision.decision === "accept") {
        accepted = { id: entry.proposalId, text: entry.text, brief: decision.buildBrief ?? "" };
        break;
      }
    }
  } catch (error) {
    return fail("triage", error instanceof Error ? error.message : "triage failed");
  }
  if (!accepted) return finish({ outcome: "dark", failure: null, darkReason: "no proposal passed triage" });
  proposalId = accepted.id;
  models.build = buildModel;

  const attempt = async (run: () => ReturnType<CrewRuntime["build"]>): Promise<{ html: string; floorJson: string } | CrewOutput> => {
    let build;
    try {
      build = await run();
    } catch (error) {
      return fail("build", error instanceof Error ? error.message : "the build failed");
    }
    sessionId = build.sessionId;
    cost += build.costCents;
    transcript = build.transcript;
    if (build.status !== "done") return fail("build", build.reason ?? "the build did not finish");
    if (!build.html || !build.floorJson) return fail("build", "The agent did not write index.html and floor.json.");
    return { html: build.html, floorJson: build.floorJson };
  };

  const brief = accepted.brief;
  const text = accepted.text;
  const shiftSlug = input.shift.slug;
  let files = await attempt(() => runtime.build({ shiftSlug, brief, proposalText: text, kitFiles }));
  if ("result" in files) return files;

  models.review = CREW_SUPPORT_MODEL;
  try {
    const first = await reviewFloor(model, brief, files.html, files.floorJson);
    cost += first.costCents;
    reviews.push(first.decision);
    if (first.decision.decision === "send-back") {
      const id = sessionId ?? "";
      const reasons = first.decision.reasons;
      files = await attempt(() => runtime.sendBack(id, reasons));
      if ("result" in files) return files;
      const second = await reviewFloor(model, brief, files.html, files.floorJson);
      cost += second.costCents;
      reviews.push(second.decision);
      if (second.decision.decision === "send-back") return fail("review", `The review agent sent the floor back twice: ${second.decision.reasons.join(" ")}`);
    }
  } catch (error) {
    return fail("review", error instanceof Error ? error.message : "review failed");
  }
  return finish({ outcome: "built", failure: null, darkReason: null }, files.html, files.floorJson);
}
```

```ts
import { assembleFloor } from "@/lib/floor-gate/assemble";
import { FloorSpec } from "@/lib/floor-gate/floor-spec";
import { runStaticGate } from "@/lib/floor-gate/static";

export type FloorCheck = { pass: boolean; findings: string[] };

/** An id a test names has to appear somewhere in the file, or that test can never pass. */
function missingIds(html: string, spec: FloorSpec): string[] {
  const findings = new Set<string>();
  for (const test of spec.tests) {
    for (const step of test.steps) {
      for (const match of step.selector.matchAll(/#([A-Za-z][\w-]*)/g)) {
        if (!html.includes(match[1])) findings.add(`floor.json: the test "${test.name}" uses ${match[0]}, but "${match[1]}" appears nowhere in index.html.`);
      }
    }
  }
  return [...findings];
}

/**
 * What run_floor_check does. It parses the agent's files and never runs them: this runs in the job
 * that holds the OpenRouter key, so the browser gate stays in the secretless verify job.
 */
export function checkFloor(html: string | null, floorJson: string | null): FloorCheck {
  const findings: string[] = [];
  if (html === null) findings.push("index.html does not exist yet. Write it with write_file.");
  if (floorJson === null) findings.push("floor.json does not exist yet. Write it with write_file.");
  if (html !== null) {
    try {
      for (const finding of runStaticGate(html)) findings.push(`${finding.rule}: ${finding.detail}`);
      assembleFloor(html, null);
    } catch (error) {
      findings.push(`index.html could not be checked: ${error instanceof Error ? error.message : "unknown error"}`);
    }
  }
  if (floorJson !== null) {
    let parsed: unknown;
    let isJson = true;
    try {
      parsed = JSON.parse(floorJson);
    } catch {
      isJson = false;
      findings.push("floor.json is not valid JSON.");
    }
    if (isJson) {
      const spec = FloorSpec.safeParse(parsed);
      if (!spec.success) for (const issue of spec.error.issues) findings.push(`floor.json: ${issue.path.join(".")}: ${issue.message}`);
      else if (html !== null) findings.push(...missingIds(html, spec.data));
    }
  }
  return { pass: findings.length === 0, findings };
}

export function checkReport(check: FloorCheck): string {
  const note = "This check reads your files and does not run them. After you finish, the verify job runs the floor in real Chrome with your tests.";
  if (check.pass) return `CHECK PASS\n${note}`;
  return ["CHECK FAIL", ...check.findings.map((finding) => `- ${finding}`), note].join("\n");
}
```

```ts
  model: string;
  capCents: number;
  transport?: Transport;
  maxTurns?: number;
  tokenCap?: number;
  deadlineMs?: number;
  sleep?: (ms: number) => Promise<void>;

  {
    name: "write_file",
    description: "Write index.html or floor.json, replacing the whole file. Those are the only two files you can write.",
    input_schema: { type: "object", additionalProperties: false, required: ["path", "content"], properties: { path: { type: "string", enum: [...WRITABLE] }, content: { type: "string" } } },

  const inside = relative(root, full);
  if (inside === "" || inside.startsWith("..") || isAbsolute(inside)) return null;
  return full;

type Conversation = {
  id: string;
  dir: string;
  system: string;
  messages: MessageParam[];
  events: RawEvent[];
  costCents: number;
  reportedCents: number;
  tokens: number;

  turns: number;
  nudges: number;
  maxTokens: number;
  shrunk: boolean;

  checkPassed: boolean;

const realSleep = (ms: number): Promise<void> => new Promise((resolve) => setTimeout(resolve, ms));

/** The build agent: a model call in a loop on the runner, with no authority. See the spec, section 5. */
export class OpenRouterRuntime implements CrewRuntime {
  private conversation: Conversation | null = null;
  private readonly now: () => number;
  private readonly sleep: (ms: number) => Promise<void>;

  constructor(private readonly client: Anthropic, private readonly config: RuntimeConfig) {
    assertAllowedModel(config.model);
    this.now = config.now ?? Date.now;
    this.sleep = config.sleep ?? realSleep;
  }

  async build(input: BuildInput): Promise<BuildResult> {
    const dir = realpathSync(mkdtempSync(join(tmpdir(), "nightcrew-build-")));

      id: randomUUID(),
      dir,
      system: [instructions, ...kit].join("\n\n"),

      reportedCents: 0,
      tokens: 0,
      conversationTokens: 0,
      turns: 0,
      nudges: 0,
      maxTokens: BUILD_MAX_TOKENS,
      shrunk: false,
      checkPassed: false,
      deadlineAt: this.now() + (this.config.deadlineMs ?? DEADLINE_MS),
    };
    return this.run(this.conversation);

  async sendBack(sessionId: string, reasons: string[]): Promise<BuildResult> {
    const conversation = this.conversation;
    if (!conversation || conversation.id !== sessionId) throw new Error("send-back for an unknown build");
    const prompt = sendBackPrompt(reasons);
    conversation.messages.push({ role: "user", content: prompt });
    conversation.events.push({ type: "user.message", processed_at: this.stamp(), content: prompt });
    conversation.checkPassed = false;
    conversation.nudges = 0;
    return this.run(conversation);
  }

  /** Removes the temp folder. The crew job calls this once the shift's output is safe. */
  dispose(): void {
    if (this.conversation) rmSync(this.conversation.dir, { recursive: true, force: true });
  }

  private stamp(): string {
    return new Date(this.now()).toISOString();
  }

  /** The stop that applies now, if any. Each is checked before a call, so no call starts past a limit. */
  private limitReached(conversation: Conversation): { status: BuildResult["status"]; reason: string } | null {
    if (this.now() > conversation.deadlineAt) return { status: "timeout", reason: "The build passed its 30-minute limit." };
    if (conversation.costCents >= this.config.capCents) return { status: "budget", reason: "spend cap reached" };
    if (conversation.tokens >= (this.config.tokenCap ?? TOKEN_CAP) || conversation.conversationTokens >= MAX_CONVERSATION_TOKENS) return { status: "budget", reason: "token cap reached" };
    const maxTurns = this.config.maxTurns ?? MAX_TURNS;
    if (conversation.turns >= maxTurns) return { status: "error", reason: `The build passed its limit of ${maxTurns} model turns.` };
    return null;
  }

  private async run(conversation: Conversation): Promise<BuildResult> {
    for (;;) {
      const limit = this.limitReached(conversation);
      if (limit) return this.finish(conversation, limit.status, limit.reason);
      // The system prompt, the tools and earlier messages never change: thinking blocks are bound to them.
      const body: OpenRouterBody = {
        model: this.config.model,
        max_tokens: conversation.maxTokens,
        system: conversation.system,
        messages: conversation.messages,
        tools: CREW_TOOLS,
        output_config: { effort: "medium" },
        provider: PINNED_PROVIDER,
        cache_control: { type: "ephemeral" },
      };
      let sent;
      try {
        sent = await sendMessage(this.client, body, { transport: this.config.transport, sleep: this.sleep });
      } catch (error) {
        const failure = classifyFailure(error);
        if (failure.kind === "too-large" && !conversation.shrunk) {
          conversation.shrunk = true;
          conversation.maxTokens = BUILD_MIN_MAX_TOKENS;
          continue;
        }
        return this.finish(conversation, "error", failure.reason);
      }
      const { message, costCents } = sent;
      const usage = message.usage;
      conversation.costCents += costCents;
      conversation.turns += 1;
      conversation.conversationTokens = usage.input_tokens + (usage.cache_creation_input_tokens ?? 0) + (usage.cache_read_input_tokens ?? 0);
      conversation.tokens += conversation.conversationTokens + usage.output_tokens;
      conversation.messages.push({ role: "assistant", content: message.content as MessageParam["content"] });
      this.recordReply(conversation, message.content);

      const uses = message.content.filter((block): block is ToolUseBlock => block.type === "tool_use");
      if (message.stop_reason === "refusal") return this.finish(conversation, "error", "The model declined this build.");
      if (uses.length > 0) {
        const cutOff = message.stop_reason === "max_tokens";
        const results: ToolResultBlockParam[] = uses.map((use) => {
          const outcome = cutOff ? { text: "Your reply hit the token limit, so nothing ran. Write a smaller file.", isError: true } : this.runTool(conversation, use);
          conversation.events.push({ type: "agent.tool_result", processed_at: this.stamp(), is_error: outcome.isError, content: outcome.text });
          return { type: "tool_result", tool_use_id: use.id, content: outcome.text, is_error: outcome.isError };
        });
        conversation.messages.push({ role: "user", content: results });
        continue;
      }
      if (message.stop_reason === "max_tokens") {
        if (conversation.nudges >= MAX_NUDGES) return this.finish(conversation, "error", "The model kept hitting its token limit.");
        conversation.nudges += 1;
        this.nudge(conversation, "Your reply hit the token limit before you used a tool. Write a smaller floor.");
        continue;
      }
      if (message.stop_reason !== "end_turn" && message.stop_reason !== "stop_sequence") return this.finish(conversation, "error", `The model stopped with ${String(message.stop_reason)}.`);
      // It says it is done. Hold it to the same check once more, because that is free.
      const check = this.check(conversation);
      if (check.pass || conversation.nudges >= MAX_NUDGES) return this.finish(conversation, "done", null);
      conversation.nudges += 1;
      this.nudge(conversation, `You are not done. Fix these and call run_floor_check again.\n${checkReport(check)}`);
    }
  }

  private nudge(conversation: Conversation, text: string): void {
    conversation.messages.push({ role: "user", content: text });
    conversation.events.push({ type: "user.message", processed_at: this.stamp(), content: text });
  }

  private recordReply(conversation: Conversation, content: ContentBlock[]): void {
    const at = this.stamp();
    for (const block of content) {
      if (block.type === "text") conversation.events.push({ type: "agent.message", processed_at: at, content: block.text });
      if (block.type === "tool_use") conversation.events.push({ type: "agent.tool_use", processed_at: at, name: block.name, input: block.input });
    }
  }

  private fileText(conversation: Conversation, name: string): string | null {
    const path = join(conversation.dir, name);
    return existsSync(path) ? readFileSync(path, "utf8") : null;
  }

  private check(conversation: Conversation) {
    return checkFloor(this.fileText(conversation, "index.html"), this.fileText(conversation, "floor.json"));
  }

  private runTool(conversation: Conversation, use: ToolUseBlock): ToolOutcome {

    if (use.name === "write_file") {
      if (typeof input.path !== "string" || !WRITABLE.includes(input.path)) return { text: `You can only write ${WRITABLE.join(" or ")}.`, isError: true };
      if (typeof input.content !== "string") return { text: "write_file needs the file's whole content as a string.", isError: true };
      if (Buffer.byteLength(input.content) > MAX_FLOOR_BYTES) return { text: `That is over the ${MAX_FLOOR_BYTES} byte limit, so nothing was written.`, isError: true };
      const target = confine(conversation.dir, input.path);
      if (!target) return { text: "That path is not allowed.", isError: true };
      mkdirSync(dirname(target), { recursive: true });
      writeFileSync(target, input.content);
      conversation.checkPassed = false;
      return { text: `Wrote ${input.path}, ${Buffer.byteLength(input.content)} bytes.`, isError: false };
    }
    if (use.name === "read_file") {
      const target = confine(conversation.dir, input.path);
      if (!target) return { text: "That path is not allowed. Paths are relative to your work folder.", isError: true };
      if (!existsSync(target) || !statSync(target).isFile()) return { text: "That file does not exist.", isError: true };
      const real = relative(conversation.dir, realpathSync(target));
      if (real.startsWith("..") || isAbsolute(real)) return { text: "That path is not allowed.", isError: true };
      const text = readFileSync(target, "utf8");
      return { text: text.length > MAX_READ_CHARS ? `${text.slice(0, MAX_READ_CHARS)}\n[cut at ${MAX_READ_CHARS} characters]` : text, isError: false };
    }
    if (use.name === "run_floor_check") {
      const check = this.check(conversation);
      conversation.checkPassed = check.pass;
      return { text: checkReport(check), isError: !check.pass };
    }
    return { text: `There is no tool called ${use.name}.`, isError: true };
  }

  private finish(conversation: Conversation, status: BuildResult["status"], reason: string | null): BuildResult {
    // A floor that already passed its check still ships if a limit stops the closing message.
    const stoppedAfterPassing = status !== "done" && conversation.checkPassed;
    const finalStatus = stoppedAfterPassing ? "done" : status;
    conversation.events.push({ type: "session.status_idle", processed_at: this.stamp(), stop_reason: { type: status === "done" ? "end_turn" : (reason ?? status) } });
    const { entries } = scrubTranscript(normaliseEvents(conversation.events));
    // crewShift adds each result's cost, so report only what this call spent.
    const costCents = conversation.costCents - conversation.reportedCents;
    conversation.reportedCents = conversation.costCents;
    return {
      sessionId: conversation.id,
      status: finalStatus,
      reason: stoppedAfterPassing ? null : reason,
      html: this.fileText(conversation, "index.html"),
      floorJson: this.fileText(conversation, "floor.json"),
      transcript: entries,
      costCents,
    };
  }
```

```ts
import type { Browser } from "playwright-core";
import { assembleFloor } from "@/lib/floor-gate/assemble";
import { gatePassed, runBrowserGate } from "@/lib/floor-gate/browser";
import { FloorSpec } from "@/lib/floor-gate/floor-spec";
import { runStaticGate } from "@/lib/floor-gate/static";
import type { Verdict } from "@/crew/lib/contracts";

export type VerifyOutput = {
  verdict: Verdict;
  assembledHtml: string | null;
  screenshots: { wide: string; narrow: string } | null;
};

/** The authoritative gate. Runs on a machine with no secrets; the publish job trusts only its verdict. */
export async function verifyFloor(rawHtml: string, floorJsonText: string, snapshotJson: string, launch: () => Promise<Browser>): Promise<VerifyOutput> {
  const findings = runStaticGate(rawHtml);
  let spec: FloorSpec | null = null;
  let floorJsonIssues: string[] = [];
  try {
    const parsed = FloorSpec.safeParse(JSON.parse(floorJsonText));
    if (parsed.success) spec = parsed.data;
    else floorJsonIssues = parsed.error.issues.map((issue) => `${issue.path.join(".")}: ${issue.message}`);
  } catch {
    floorJsonIssues = ["floor.json is not valid JSON"];
  }
  const base = { static: findings, floorJsonIssues, violations: [], errors: [], tests: [], floorHash: null, scriptHashes: null };
  if (!spec) return { verdict: { ...base, pass: false }, assembledHtml: null, screenshots: null };

  let assembled;
  try {
    assembled = assembleFloor(rawHtml, snapshotJson);
  } catch (error) {
    const message = error instanceof Error ? error.message : "assembly failed";
    return { verdict: { ...base, pass: false, errors: [message] }, assembledHtml: null, screenshots: null };
  }
  const result = await runBrowserGate({ html: assembled.html, scriptHashes: assembled.scriptHashes, spec, launch, screenshots: true });
  const pass = findings.length === 0 && gatePassed(result);
  return {
    verdict: {
      pass,
      static: findings,
      floorJsonIssues,
      violations: result.violations,
      errors: result.errors,
      tests: result.tests,
      floorHash: assembled.hash,
      scriptHashes: assembled.scriptHashes,
    },
    assembledHtml: assembled.html,
    screenshots: result.screenshots,
  };
}
```

```ts
import { canonicalJson } from "@/lib/canonical";
import { keccakText } from "@/lib/hash";

export type ReceiptOutcome = "shipped" | "failed" | "dark";
export type ReceiptFailure = { stage: "close" | "triage" | "build" | "review" | "verify" | "publish"; reason: string };

/** Field names follow Vibestarter's Origin Capsule where they overlap. */
export type ReceiptBundle = {
  standard: "nightcrew-receipt/1";
  shift: { slug: string; label: "scheduled" | "off-schedule" | "rehearsal"; closes_at: string; started_at: string; finished_at: string; snapshot_block: string | null };
  proposal: { id: number; text: string; author: string; seeded: boolean } | null;
  outcome: ReceiptOutcome;
  failure: ReceiptFailure | null;
  dark_reason: string | null;
  agent_identity: "night-crew";
  agent_tool: "Claude Managed Agents";
  model_provider: "Anthropic";
  models: { triage: string; build: string | null; review: string | null };
  founder_address: string;
  tally_hash: string;
  triage_hash: string;
  transcript_hash: string | null;
  floor_hash: string | null;
  diff_hash: string | null;
  tests: { passed: number; total: number; hash: string } | null;
  cost_cents: number;
  timestamp: string;
};

export type ReceiptParts = {
  tally: unknown;
  triage: unknown;
  transcript: unknown | null;
  floorHtml: string | null;
  diff: string | null;
  tests: { name: string; pass: boolean; detail: string | null }[] | null;
};

export type BuiltReceipt = {
  bundle: ReceiptBundle;
  bundleText: string;
  receiptHash: `0x${string}`;
  /** Name to exact bytes, served at /api/receipts/[slug]/[part] so anyone can re-hash them. */
  parts: Record<string, string>;
};

export function buildReceipt(fields: Omit<ReceiptBundle, "standard" | "agent_identity" | "agent_tool" | "model_provider" | "tally_hash" | "triage_hash" | "transcript_hash" | "floor_hash" | "diff_hash" | "tests">, parts: ReceiptParts): BuiltReceipt {
  const out: Record<string, string> = {
    "tally.json": canonicalJson(parts.tally),
    "triage.json": canonicalJson(parts.triage),
  };
  if (parts.transcript !== null) out["transcript.json"] = canonicalJson(parts.transcript);
  if (parts.floorHtml !== null) out["floor.html"] = parts.floorHtml;
  if (parts.diff !== null) out["diff.patch"] = parts.diff;
  if (parts.tests !== null) out["tests.json"] = canonicalJson(parts.tests);

  const hashOf = (name: string): string | null => (out[name] === undefined ? null : keccakText(out[name]));
  const bundle: ReceiptBundle = {
    ...fields,
    standard: "nightcrew-receipt/1",
    agent_identity: "night-crew",
    agent_tool: "Claude Managed Agents",
    model_provider: "Anthropic",
    tally_hash: keccakText(out["tally.json"]),
    triage_hash: keccakText(out["triage.json"]),
    transcript_hash: hashOf("transcript.json"),
    floor_hash: hashOf("floor.html"),
    diff_hash: hashOf("diff.patch"),
    tests: parts.tests === null ? null : { passed: parts.tests.filter((test) => test.pass).length, total: parts.tests.length, hash: keccakText(out["tests.json"]) },
  };
  const bundleText = canonicalJson(bundle);
  out["bundle.json"] = bundleText;
  return { bundle, bundleText, receiptHash: keccakText(bundleText), parts: out };
}
```

```ts
export function receiptSchemaUid(): Hex {
  return keccak256(encodePacked(["string", "address", "bool"], [RECEIPT_SCHEMA, zeroAddress, false]));
}

export type AttestationFields = {
  receiptHash: string;
  shift: string;
  outcome: string;
  transcriptHash: string | null;
  floorHash: string | null;
  agentIdentity: string;
  modelProvider: string;
  founder: string;
  costCents: number;
};

export function encodeAttestationData(fields: AttestationFields): Hex {
  return encodeAbiParameters(parseAbiParameters(RECEIPT_SCHEMA), [
    fields.receiptHash as Hex,
    fields.shift,
    fields.outcome,
    (fields.transcriptHash ?? zeroHash) as Hex,
    (fields.floorHash ?? zeroHash) as Hex,
    fields.agentIdentity,
    fields.modelProvider,
    fields.founder as Hex,
    fields.costCents,
  ]);
}

export type AttestationResult = { uid: string; tx: string };
export type Attester = (fields: AttestationFields) => Promise<AttestationResult>;

export function viemAttester(privateKey: Hex, schemaUid: Hex, rpcUrl: string = baseRpcUrl()): Attester {
  const account = privateKeyToAccount(privateKey);
  const publicClient = createPublicClient({ chain: base, transport: http(rpcUrl) });
  const walletClient = createWalletClient({ account, chain: base, transport: http(rpcUrl) });
  return async (fields) => {
    const tx = await walletClient.writeContract({
      address: EAS_ADDRESS,
      abi: EAS_ABI,
      functionName: "attest",
      args: [{ schema: schemaUid, data: { recipient: zeroAddress, expirationTime: 0n, revocable: false, refUID: zeroHash, data: encodeAttestationData(fields), value: 0n } }],
    });
    const receipt = await publicClient.waitForTransactionReceipt({ hash: tx });
    if (receipt.status !== "success") throw new Error(`attestation reverted: ${tx}`);
    const [event] = parseEventLogs({ abi: EAS_ABI, eventName: "Attested", logs: receipt.logs });
    if (!event) throw new Error(`no Attested event in ${tx}`);
    return { uid: event.args.uid, tx };
  };
}

/** A key pasted into a secret can pick up spaces or lose its 0x. */
export function normaliseKey(raw: string): Hex {
  const key = raw.trim();
  return (key.startsWith("0x") ? key : `0x${key}`) as Hex;
}

/** Null when a value is missing or unusable: the shift still publishes, with its attestation pending. Never logs the key. */
export function attesterFromEnv(env: Record<string, string | undefined>, rpcUrl?: string): Attester | null {
  const key = env.ATTESTER_PRIVATE_KEY?.trim();
  const schema = env.EAS_SCHEMA_UID?.trim();
  if (!key || !schema) return null;
  try {
    return viemAttester(normaliseKey(key), schema as Hex, rpcUrl);
  } catch {
    console.error("ATTESTER_PRIVATE_KEY is not a valid key, so attestation stays pending.");
    return null;
  }
}

export { SCHEMA_REGISTRY_ADDRESS };
```

```ts
/** Everything publish reads from other jobs, as untrusted text. Any of it may be missing. */
export type PublishWork = {
  shiftInput: string;
  crewResult: string | null;
  transcript: string | null;
  floorHtml: string | null;
  floorJson: string | null;
  verdict: string | null;
  screenshotWide: string | null;
  screenshotNarrow: string | null;

export type PublishDeps = {
  db: Db;
  attester: Attester | null;
  work: PublishWork;
  template: string;
  founder: string;
  now: Date;

export type PublishResult = { outcome: ReceiptOutcome; receiptHash: string; attested: number; pending: number; alreadyPublished: boolean };

function safeParse<T>(schema: { safeParse(value: unknown): { success: true; data: T } | { success: false } }, text: string | null): T | null {
  if (text === null) return null;
  try {
    const result = schema.safeParse(JSON.parse(text));
    return result.success ? result.data : null;
  } catch {
    return null;
  }
}

async function attestPending(db: Db, attester: Attester | null): Promise<{ attested: number; pending: number }> {
  const rows = await db.query<{ shift_id: number; bundle: string; receipt_hash: string }>(
    "select shift_id::int as shift_id, bundle, receipt_hash from receipts where attestation_status = 'pending' order by shift_id",
  );
  let attested = 0;
  for (const row of rows.rows) {
    if (!attester) continue;
    const bundle = JSON.parse(row.bundle) as ReceiptBundle;
    try {
      const { uid, tx } = await attester({
        receiptHash: row.receipt_hash,
        shift: bundle.shift.slug,
        outcome: bundle.outcome,
        transcriptHash: bundle.transcript_hash,
        floorHash: bundle.floor_hash,
        agentIdentity: bundle.agent_identity,
        modelProvider: bundle.model_provider,
        founder: bundle.founder_address,
        costCents: bundle.cost_cents,
      });
      await db.query("update receipts set attestation_status = 'attested', attestation_uid = $1, attestation_tx = $2 where shift_id = $3", [uid, tx, row.shift_id]);
      attested += 1;
    } catch (error) {
      console.error(`attestation failed for shift ${bundle.shift.slug}: ${error instanceof Error ? error.message : String(error)}`);
    }
  }
  return { attested, pending: rows.rows.length - attested };
}

export async function publishShift({ db, attester, work, template, founder, now }: PublishDeps): Promise<PublishResult> {
  const input = ShiftInput.parse(JSON.parse(work.shiftInput));
  const shiftId = input.shift.id;

  const existing = await db.query<{ receipt_hash: string; bundle: string }>("select receipt_hash, bundle from receipts where shift_id = $1", [shiftId]);
  if (existing.rows[0]) {
    const counts = await attestPending(db, attester);
    const bundle = JSON.parse(existing.rows[0].bundle) as ReceiptBundle;
    return { outcome: bundle.outcome, receiptHash: existing.rows[0].receipt_hash, ...counts, alreadyPublished: true };
  }

  const crew = safeParse(CrewResult, work.crewResult);
  const transcript = safeParse(Transcript, work.transcript);
  const verdict = safeParse(Verdict, work.verdict);
  const spec = safeParse(FloorSpec, work.floorJson);

  let outcome: ReceiptOutcome;
  let failure: ReceiptFailure | null = null;
  let darkReason: string | null = null;
  let assembled: ReturnType<typeof assembleFloor> | null = null;

  if (input.failure) {
    outcome = "failed";
    failure = input.failure;
  } else if (input.darkReason) {
    outcome = "dark";
    darkReason = input.darkReason;
  } else if (!crew) {
    outcome = "failed";
    failure = { stage: "build", reason: "The crew job did not finish." };
  } else if (crew.outcome === "dark") {
    outcome = "dark";
    darkReason = crew.darkReason;
  } else if (crew.outcome === "failed") {
    outcome = "failed";
    failure = crew.failure ?? { stage: "build", reason: "The crew job failed." };
  } else if (!verdict) {
    outcome = "failed";
    failure = { stage: "verify", reason: "The verify job did not finish." };
  } else if (!verdict.pass) {
    outcome = "failed";
    const details = [...verdict.static.map((item) => item.detail), ...verdict.violations.map((item) => `${item.kind}: ${item.detail}`), ...verdict.errors, ...verdict.floorJsonIssues, ...verdict.tests.filter((test) => !test.pass).map((test) => `test "${test.name}" failed`)];
    failure = { stage: "verify", reason: `The gate rejected the floor: ${details.slice(0, 5).join("; ")}`.slice(0, 500) };
  } else if (!work.floorHtml || !spec) {
    outcome = "failed";
    failure = { stage: "publish", reason: "The floor files are missing or malformed." };
  } else {
    assembled = assembleFloor(work.floorHtml, input.snapshot);
    if (assembled.hash !== verdict.floorHash) {
      outcome = "failed";
      failure = { stage: "publish", reason: "The assembled floor does not match what the gate verified." };
      assembled = null;
    } else {
      outcome = "shipped";
    }
  }

  const proposalEntry = crew?.proposalId != null ? input.tally?.entries.find((entry) => entry.proposalId === crew.proposalId) : undefined;
  const diff = outcome === "shipped" && work.floorHtml ? createTwoFilesPatch("blank-floor.html", "floor.html", template.replace(/\r\n/g, "\n"), work.floorHtml.replace(/\r\n/g, "\n")) : null;
  const costCents = crew?.costCents ?? 0;

  const receipt = buildReceipt(
    {
      shift: { slug: input.shift.slug, label: input.shift.label, closes_at: input.shift.closesAt, started_at: input.shift.startedAt, finished_at: now.toISOString(), snapshot_block: input.tally?.snapshotBlock ?? null },
      proposal: proposalEntry ? { id: proposalEntry.proposalId, text: proposalEntry.text, author: proposalEntry.author, seeded: proposalEntry.seeded } : null,
      outcome,
      failure,
      dark_reason: darkReason,
      models: crew?.models ?? { triage: CREW_SUPPORT_MODEL, build: null, review: null },
      founder_address: founder.toLowerCase(),
      cost_cents: costCents,
      timestamp: now.toISOString(),
    },
    {
      tally: input.tally ?? {},
      triage: crew?.triage ?? [],
      transcript: transcript ?? null,
      floorHtml: assembled?.html ?? null,
      diff,
      tests: verdict?.tests ?? null,
    },
  );

  const statements: Statement[] = [
    {
      sql: "update shifts set status = $2, finished_at = $3, snapshot_block = $4, tally = $5, build_model = $6, cost_cents = $7, failure_stage = $8, failure_reason = $9 where id = $1",
      params: [shiftId, outcome, now.toISOString(), input.tally?.snapshotBlock ?? null, JSON.stringify(input.tally ?? {}), crew?.models.build ?? null, costCents, failure?.stage ?? null, failure?.reason ?? darkReason],
    },
    {
      sql: "insert into receipts (shift_id, bundle, receipt_hash) values ($1, $2, $3)",
      params: [shiftId, receipt.bundleText, receipt.receiptHash],
    },
    ...Object.entries(receipt.parts).map(([part, content]) => ({ sql: "insert into receipt_parts (shift_id, part, content) values ($1, $2, $3)", params: [shiftId, part, content] })),
  ];

  if (outcome === "shipped" && assembled && spec) {
    statements.push({
      sql: "insert into floors (shift_id, name, one_liner, how_to_use, html, hash, script_hashes, screenshot_wide, screenshot_narrow) values ($1, $2, $3, $4, $5, $6, $7, $8, $9)",
      params: [shiftId, spec.name, spec.oneLiner, spec.howToUse, assembled.html, assembled.hash, assembled.scriptHashes, work.screenshotWide, work.screenshotNarrow],
    });
  }
  for (const decision of crew?.triage ?? []) {
    if (decision.decision === "reject") {
      statements.push({ sql: "update proposals set status = 'rejected', status_reason = $2, status_category = $3 where id = $1 and status = 'open'", params: [decision.proposalId, decision.reason, decision.category] });
    }
  }
  if (crew?.proposalId != null && input.shift.label !== "rehearsal") {
    if (outcome === "shipped") {
      statements.push({ sql: "update proposals set status = 'built', built_in = $2 where id = $1", params: [crew.proposalId, shiftId] });
    } else if (outcome === "failed") {
      statements.push({ sql: "update proposals set failed_attempts = failed_attempts + 1 where id = $1", params: [crew.proposalId] });
    }
  }
  await db.batch(statements);

  const counts = await attestPending(db, attester);
  return { outcome, receiptHash: receipt.receiptHash, ...counts, alreadyPublished: false };
}
```

```ts
import Anthropic, { APIError } from "@anthropic-ai/sdk";
import type { Message, MessageCreateParamsNonStreaming } from "@anthropic-ai/sdk/resources/messages/messages";

/** OpenRouter's Anthropic-compatible endpoint. The SDK adds /v1/messages. */
export const OPENROUTER_BASE_URL = "https://openrouter.ai/api";
export const SONNET_SLUG = "anthropic/claude-sonnet-5.5";
export const HAIKU_SLUG = "anthropic/claude-haiku-5.5";
/** Thinking counts toward this. A request's up-front estimate on a low balance depends on it too. */
export const BUILD_MAX_TOKENS = 16_000;
/** Where the build drops to, once, if a request is too large for the in-flight budget. */
export const BUILD_MIN_MAX_TOKENS = 8_000;

const ALLOWED_MODELS: readonly string[] = [SONNET_SLUG, HAIKU_SLUG];
const MAX_WAITS = 3;
const MAX_WAIT_MS = 60_000;
const COST_LOOKUP_ATTEMPTS = 6;

/** Every request goes to Anthropic's own endpoint, so behaviour never changes with routing. */
export const PINNED_PROVIDER = { only: ["anthropic"], allow_fallbacks: false };

export type Transport = "non-stream" | "stream";

export type OpenRouterBody = MessageCreateParamsNonStreaming & {
  provider: { only: string[]; allow_fallbacks: boolean; require_parameters?: boolean };
  cache_control?: { type: "ephemeral" };
};

export type SentMessage = { message: Message; costCents: number; costSource: "response" | "lookup" };
export type SendOptions = { transport?: Transport; sleep?: (ms: number) => Promise<void> };

export type FailureKind = "in-flight" | "too-large" | "no-credit" | "blocked" | "transient" | "other";
export type ProviderFailure = { kind: FailureKind; reason: string; retryAfterMs: number | null };

/** A failure in words fit for a public receipt. */
export class ProviderError extends Error {}

const realSleep = (ms: number): Promise<void> => new Promise((resolve) => setTimeout(resolve, ms));

export function openRouterClient(apiKey: string, options: { fetch?: typeof fetch } = {}): Anthropic {
  return new Anthropic({
    baseURL: OPENROUTER_BASE_URL,
    apiKey: null,
    authToken: apiKey,
    timeout: 6 * 60 * 1000,
    // sendMessage retries only an HTTP error, which is not billed. The SDK would also retry a timed-out call, which may be.
    maxRetries: 0,
    ...(options.fetch ? { fetch: options.fetch } : {}),
  });
}

/** Exact slugs only. A suffix such as :online would add web search, which the crew must never have. */
export function assertAllowedModel(model: string): string {
  if (!ALLOWED_MODELS.includes(model)) throw new Error(`${model} is not an allowed crew model. Use ${ALLOWED_MODELS.join(" or ")}.`);
  return model;
}

export function parseTransport(value: string | undefined): Transport {
  if (!value || value === "non-stream") return "non-stream";
  if (value === "stream") return "stream";
  throw new Error(`CREW_BUILD_TRANSPORT must be non-stream or stream, not ${value}`);
}

const retryAfterMs = (headers: Headers | undefined): number | null => {
  const seconds = Number(headers?.get("retry-after"));
  return Number.isFinite(seconds) && seconds > 0 ? seconds * 1000 : null;
};

/** The named fields only. A moderation error echoes the flagged text, so the body as a whole is hostile. */
function errorFields(error: APIError): { reason: string; source: string; message: string } {
  const body = (error.error ?? {}) as { error?: unknown };
  const inner = (body.error ?? body) as { message?: unknown; metadata?: { reason?: unknown; limit_source?: unknown } };
  return { reason: String(inner.metadata?.reason ?? ""), source: String(inner.metadata?.limit_source ?? ""), message: String(inner.message ?? "") };
}

/** Sorts a failed request by what to do next. Never repeats a response body in public words. */
export function classifyFailure(error: unknown): ProviderFailure {
  if (error instanceof ProviderError) return { kind: "other", reason: error.message, retryAfterMs: null };
  if (!(error instanceof APIError)) return { kind: "other", reason: "The model request failed before a reply arrived.", retryAfterMs: null };
  const { reason, source, message } = errorFields(error);
  const wait = retryAfterMs(error.headers);
  if (error.status === 402) {
    if (reason === "in_flight_budget_exhausted" || source === "openrouter_in_flight_budget") return { kind: "in-flight", reason: "OpenRouter's in-flight budget stayed full.", retryAfterMs: wait };
    if (reason === "weight_exceeds_budget") return { kind: "too-large", reason: "One request was larger than OpenRouter's in-flight budget allows at this balance.", retryAfterMs: null };
    if (source === "openrouter_key_limit") return { kind: "no-credit", reason: "The OpenRouter key's credit limit is used up.", retryAfterMs: null };
    // The Messages endpoint wraps errors in Anthropic's shape, which may drop OpenRouter's metadata.
    if (/in[- ]flight/i.test(message)) {
      return wait === null ? { kind: "too-large", reason: "One request was larger than OpenRouter's in-flight budget allows at this balance.", retryAfterMs: null } : { kind: "in-flight", reason: "OpenRouter's in-flight budget stayed full.", retryAfterMs: wait };
    }
    return { kind: "no-credit", reason: "OpenRouter credit ran out.", retryAfterMs: null };
  }
  if (error.status === 403) {
    if (/budget|key_limit/.test(source) || /\$[\d.]+ of \$[\d.]+ used/.test(message)) return { kind: "no-credit", reason: "The OpenRouter key's budget is used up.", retryAfterMs: null };
    return { kind: "blocked", reason: "A content filter or a model refusal blocked the request.", retryAfterMs: null };
  }
  if (error.status === 401) return { kind: "other", reason: "OpenRouter refused the key.", retryAfterMs: null };
  if (error.status !== undefined && [429, 500, 502, 503, 529].includes(error.status)) return { kind: "transient", reason: `The model request failed with status ${error.status}.`, retryAfterMs: wait };
  return { kind: "other", reason: `The model request failed with status ${error.status ?? "unknown"}.`, retryAfterMs: null };
}

const readCost = (usage: unknown): number | null => {
  const cost = (usage as { cost?: unknown } | null | undefined)?.cost;
  return typeof cost === "number" && Number.isFinite(cost) ? cost : null;
};

/** The published stream schema has no cost, so a stream falls back to OpenRouter's generation record. */
async function lookupCost(client: Anthropic, id: string, sleep: (ms: number) => Promise<void>): Promise<number> {
  for (let attempt = 0; attempt < COST_LOOKUP_ATTEMPTS; attempt += 1) {
    if (attempt > 0) await sleep(2000);
    try {
      const found = await client.get<{ data?: { total_cost?: unknown } }>("/v1/generation", { query: { id } });
      const total = found.data?.total_cost;
      if (typeof total === "number" && Number.isFinite(total)) return total;
    } catch (error) {
      if (!(error instanceof APIError) || error.status !== 404) throw error;
    }
  }
  throw new ProviderError("OpenRouter reported no cost for a streamed call, so the run stopped rather than spend blind.");
}

async function sendOnce(client: Anthropic, body: OpenRouterBody, transport: Transport, sleep: (ms: number) => Promise<void>): Promise<SentMessage> {
  if (transport === "non-stream") {
    const message = await client.messages.create(body);
    if (!Array.isArray(message?.content)) throw new ProviderError("OpenRouter returned an error instead of a message.");
    const cost = readCost(message.usage);
    if (cost === null) throw new ProviderError("OpenRouter reported no cost for a call, so the run stopped rather than spend blind.");
    return { message, costCents: cost * 100, costSource: "response" };
  }
  const stream = client.messages.stream(body);
  const seen: { cost: number | null } = { cost: null };
  stream.on("streamEvent", (event) => {
    const usage = event.type === "message_delta" ? event.usage : event.type === "message_start" ? event.message.usage : null;
    seen.cost = readCost(usage) ?? seen.cost;
  });
  const message = await stream.finalMessage();
  if (seen.cost !== null) return { message, costCents: seen.cost * 100, costSource: "response" };
  const id = stream.response?.headers.get("x-generation-id") ?? message.id;
  return { message, costCents: (await lookupCost(client, id, sleep)) * 100, costSource: "lookup" };
}

/** One model call, priced from what OpenRouter reports. Waits out a full in-flight budget or a passing HTTP error. */
export async function sendMessage(client: Anthropic, body: OpenRouterBody, options: SendOptions = {}): Promise<SentMessage> {
  const sleep = options.sleep ?? realSleep;
  for (let attempt = 0; ; attempt += 1) {
    try {
      return await sendOnce(client, body, options.transport ?? "non-stream", sleep);
    } catch (error) {
      const failure = classifyFailure(error);
      if ((failure.kind !== "in-flight" && failure.kind !== "transient") || attempt >= MAX_WAITS) throw error;
      await sleep(Math.min(failure.retryAfterMs ?? 5000, MAX_WAIT_MS));
    }
  }
}
```

```ts
      method: init?.method ?? "GET",
      headers: new Headers(init?.headers),
      body: raw ? (JSON.parse(raw) as Record<string, unknown>) : {},
    };
    requests.push(request);
    const reply = queue ? queue.shift() : (script as (request: FakeRequest) => FakeReply)(request);
    if (!reply) throw new Error(`no scripted reply left for ${request.url}`);
    if (reply.events) {
      const text = reply.events.map((item) => `event: ${item.event}\ndata: ${JSON.stringify(item.data)}\n\n`).join("");
      return new Response(text, { status: 200, headers: { "content-type": "text/event-stream", ...reply.headers } });
    }
    return new Response(JSON.stringify(reply.body ?? {}), { status: reply.status ?? 200, headers: { "content-type": "application/json", ...reply.headers } });
  }) as typeof fetch;
  return { fetch: fakeFetch, requests };
}
```

```ts
import type Anthropic from "@anthropic-ai/sdk";
import { classifyFailure, PINNED_PROVIDER, sendMessage, type OpenRouterBody } from "@/crew/lib/openrouter";

export type JsonSchema = Record<string, unknown>;

export type StructuredCall = {
  model: string;
  system: string;
  user: string;
  schema: JsonSchema;
  maxTokens: number;
};

export type StructuredResult = {
  /** Null when the model refused or returned nothing parseable. */
  value: unknown;
  refused: boolean;
  costCents: number;
};

/** A single call with no working tools, whose reply is constrained to a schema. */
export type StructuredModel = (call: StructuredCall) => Promise<StructuredResult>;

/** forced-tool is the fallback if the probe finds structured output unsupported. Callers validate either way. */
export type StructuredMode = "json-schema" | "forced-tool";

const ANSWER_TOOL = "answer";

export function parseStructuredMode(value: string | undefined): StructuredMode {
  if (!value || value === "json-schema") return "json-schema";
  if (value === "forced-tool") return "forced-tool";
  throw new Error(`CREW_STRUCTURED_MODE must be json-schema or forced-tool, not ${value}`);
}

function structuredBody({ model, system, user, schema, maxTokens }: StructuredCall, mode: StructuredMode): OpenRouterBody {
  const base = {
    model,
    max_tokens: maxTokens,
    system,
    messages: [{ role: "user" as const, content: user }],
    provider: { ...PINNED_PROVIDER, require_parameters: true },
  };
  if (mode === "json-schema") return { ...base, output_config: { effort: "low", format: { type: "json_schema", schema } } };
  // A forced tool only shapes the reply. Nothing runs it. Claude Sonnet 5.5 rejects this, so it stays on Haiku.
  return {
    ...base,
    output_config: { effort: "low" },
    tools: [{ name: ANSWER_TOOL, description: "Give your answer.", input_schema: { type: "object", ...schema } }],
    tool_choice: { type: "tool", name: ANSWER_TOOL },
  };
}

export function openRouterStructuredModel(client: Anthropic, mode: StructuredMode = "json-schema"): StructuredModel {
  return async (call) => {
    let sent;
    try {
      sent = await sendMessage(client, structuredBody(call, mode));
    } catch (error) {
      const failure = classifyFailure(error);
      if (failure.kind === "blocked") return { value: null, refused: true, costCents: 0 };
      throw new Error(failure.reason);
    }
    const { message, costCents } = sent;
    if (message.stop_reason === "refusal") return { value: null, refused: true, costCents };
    if (mode === "forced-tool") {
      const block = message.content.find((item) => item.type === "tool_use" && item.name === ANSWER_TOOL);
      return { value: block?.type === "tool_use" ? block.input : null, refused: false, costCents };
    }
    const text = message.content.flatMap((block) => (block.type === "text" ? [block.text] : [])).join("");
    try {
      return { value: JSON.parse(text), refused: false, costCents };
    } catch {
      return { value: null, refused: false, costCents };
    }
  };
}
```

```ts
import type { JsonSchema, StructuredModel } from "@/crew/lib/model";

export const CREW_SUPPORT_MODEL = HAIKU_SLUG;

const TRIAGE_SCHEMA: JsonSchema = {
  type: "object",
  additionalProperties: false,
  required: ["decision", "category", "reason", "buildBrief"],
  properties: {
    decision: { type: "string", enum: ["accept", "reject"] },
    category: { type: "string", enum: TriageCategory.options },
    reason: { type: "string" },
    buildBrief: { type: ["string", "null"] },
  },
};

export function triageSystemPrompt(): string {
  return readFileSync(join(process.cwd(), "crew", "prompts", "triage.md"), "utf8");
}

/** JSON-encodes the proposal, so it cannot close the block it sits in. */
export function triageUserMessage(text: string): string {
  return `<proposal>${JSON.stringify(text).replace(/</g, "\\u003c")}</proposal>`;
}

export type TriageOutcome = { decision: TriageDecision; costCents: number };

export async function triageProposal(model: StructuredModel, proposalId: number, text: string): Promise<TriageOutcome> {
  const result = await model({ model: CREW_SUPPORT_MODEL, system: triageSystemPrompt(), user: triageUserMessage(text), schema: TRIAGE_SCHEMA, maxTokens: 2000 });
  const reject = (category: TriageCategory, reason: string): TriageOutcome => ({
    decision: { proposalId, decision: "reject", category, reason, buildBrief: null },
    costCents: result.costCents,
  });
  if (result.refused) return reject("model-refusal", "The model declined to assess this proposal.");
  const value = result.value as { reason?: unknown; buildBrief?: unknown } | null;
  const parsed = TriageDecision.safeParse({
    ...(value ?? {}),
    proposalId,
    reason: typeof value?.reason === "string" ? value.reason.slice(0, 200) : value?.reason,
    buildBrief: typeof value?.buildBrief === "string" ? value.buildBrief.slice(0, 800) : value?.buildBrief,
  });
  if (!parsed.success) throw new Error("triage returned an invalid decision");
  const decision = parsed.data;
  if (decision.decision === "accept" && (!decision.buildBrief || decision.category !== "ok")) throw new Error("triage accepted without a brief");
  return { decision, costCents: result.costCents };
}
```

```ts
import type { TranscriptEntry } from "@/crew/lib/contracts";

const MAX_ENTRY_CHARS = 20_000;
const KEY_PATTERNS = [/sk-or-[A-Za-z0-9_-]{8,}/g, /sk-ant-[A-Za-z0-9_-]{8,}/g, /\b0x[a-fA-F0-9]{64}\b/g];

export type RawEvent = { type: string; [key: string]: unknown };

function blocksToText(content: unknown): string {
  if (typeof content === "string") return content;
  if (!Array.isArray(content)) return "";
  return content.map((block: { type?: string; text?: string }) => (block?.type === "text" ? (block.text ?? "") : `[${block?.type ?? "block"}]`)).join("\n");
}

function entry(event: RawEvent, kind: TranscriptEntry["kind"], text: string, tool: string | null = null, isError = false): TranscriptEntry {
  const truncated = text.length > MAX_ENTRY_CHARS;
  return {
    at: String(event.processed_at ?? event.created_at ?? ""),
    kind,
    tool,
    text: truncated ? `${text.slice(0, MAX_ENTRY_CHARS)}\n[truncated]` : text,
    truncated,
    isError,
  };
}

/** Turns session events into the public transcript. Thinking has no content, so it is left out. */
export function normaliseEvents(events: RawEvent[]): TranscriptEntry[] {
  const out: TranscriptEntry[] = [];
  for (const event of events) {
    switch (event.type) {
      case "user.message":
        out.push(entry(event, "prompt", blocksToText(event.content)));
        break;
      case "agent.message":
        out.push(entry(event, "message", blocksToText(event.content)));
        break;
      case "agent.tool_use":
        out.push(entry(event, "tool_use", JSON.stringify(event.input ?? {}), String(event.name ?? "tool")));
        break;
      case "agent.tool_result":
        out.push(entry(event, "tool_result", blocksToText(event.content), null, Boolean(event.is_error)));
        break;
      case "session.error":
        out.push(entry(event, "error", JSON.stringify(event.error ?? event)));
        break;
      case "session.status_idle":
        out.push(entry(event, "status", `idle: ${String((event.stop_reason as { type?: string } | undefined)?.type ?? "unknown")}`));
        break;
      default:
        break;
    }
  }
  return out;
}

/** Removes anything key-shaped. The agent's tools can reach no secrets, so a match is a defect worth seeing. */
export function scrubTranscript(entries: TranscriptEntry[]): { entries: TranscriptEntry[]; removed: number } {
  let removed = 0;
  const clean = entries.map((item) => {
    let text = item.text;
    for (const pattern of KEY_PATTERNS) {
      text = text.replace(pattern, () => {
        removed += 1;
        return "[removed]";
      });
    }
    return { ...item, text };
  });
  if (removed === 0) return { entries: clean, removed };
  const marker: TranscriptEntry = {
    at: clean[clean.length - 1]?.at ?? "",
    kind: "status",
    tool: null,
    text: `Removed ${removed} key-shaped string${removed === 1 ? "" : "s"} from this transcript. The agent's tools can reach no secrets, so this is a defect worth seeing.`,
    truncated: false,
    isError: true,
  };
  return { entries: [...clean, marker], removed };
}
```
