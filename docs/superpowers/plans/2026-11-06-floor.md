# Floor sandbox and gate Implementation Plan

**Goal:** Make it safe to run AI-written code next to a visitor's wallet, and prove it against a set of hostile floors.
**Architecture:** A floor is one HTML file. Assembly injects a wallet guard first and wraps the floor's script so window and wallet globals are shadowed. It is served under a CSP with hash-pinned scripts and the sandbox directive. A browser gate runs it in real Chrome with a fake injected wallet. The static rules are split: core rules guard self-navigation, extended rules give clearer errors and are cut candidate 4.
**Tech Stack:** parse5 8.1.0, acorn 8.19.0, acorn-walk 8.3.5, playwright-core 1.65.0, Chrome from CHROME_PATH
**Source spec:** docs/superpowers/specs/2026-11-06-nightcrew-design.md

Task references use `<plan>:<task>`, across all six plans, in build order: `foundation`, `floor`, `pipeline`, `votes`, `frontend`, `launch`. The code here passed each task's gate in a reference build on 2026-11-06. Tasks marked as cut candidates can be skipped at the Sat 7 Nov 12:00 UTC checkpoint, through the architect, never by hand. Gates need the installed dependencies: a fresh worktree runs `npm ci` first, with a real `node_modules` folder and never a link to another one.

**Amended 2026-11-06: the crew runs on OpenRouter, not Managed Agents.** Tasks already built keep their original text, so some still name Managed Agents: `foundation:7`, `pipeline:14` and `pipeline:15` amend them. Tasks not yet built were edited in place, and every task id is unchanged.

 * reference to the real window. Floor code gets `__ncw`, a restricted proxy, through the wrapper
 * in assemble.ts. Stored as a string so it can be hashed for the CSP.
 *

 * this script finishes is taken here, up front: nothing below looks a method up at run time. The
 * lookup tables have no prototype, so a name like `valueOf` is not on the allowlist by inheritance.
 */

  var eventPath = Event.prototype.composedPath;
  var eventType = describe(Event.prototype, "type").get;
  var mapGet = WeakMap.prototype.get;
  var mapSet = WeakMap.prototype.set;

  var startRepeat = window.setInterval;

    try { define(window, n, { get: function () { return undefined; }, set: function () {}, configurable: false }); } catch (e) {}
  });

  apply(realAdd, window, ["eip6963:announceProvider", swallow, true]);
  apply(realAdd, window, ["eip6963:requestProvider", swallow, true]);

import { parse as parseJs } from "acorn";
import { defaultTreeAdapter, html as htmlNs, parse, serialize, type DefaultTreeAdapterTypes } from "parse5";
import { keccakText, cspScriptHash } from "@/lib/hash";
import { WALLET_GUARD } from "@/lib/floor-gate/guard";

type TreeNode = DefaultTreeAdapterTypes.Node;
type ElementNode = DefaultTreeAdapterTypes.Element;

const UNDEFINED_NAMES = ["self", "globalThis", "top", "parent", "frames", "opener", "location", "history", "chrome", "external", "ethereum", "web3", "solana", "phantom", "fetch", "XMLHttpRequest", "WebSocket", "EventSource", "postMessage", "localStorage", "sessionStorage", "indexedDB", "open", "navigation", "frameElement", "caches", "cookieStore", "documentPictureInPicture"];
const SAFE_NAMES = ["addEventListener", "removeEventListener", "setTimeout", "setInterval", "clearTimeout", "clearInterval", "requestAnimationFrame", "cancelAnimationFrame", "matchMedia", "getComputedStyle"];

const PARAMS = ["window", ...UNDEFINED_NAMES, ...SAFE_NAMES].join(",");
const ARGS = ["__ncw", ...UNDEFINED_NAMES.map(() => "void 0"), ...SAFE_NAMES.map((name) => `__ncw.${name}`)].join(",");

/**
 * Wraps floor code at global scope, so no closure can leak the real window. The inner block lets
 * a floor declare its own `top` or `location` with let or const without clashing with a parameter.
 * Refuses anything that is not a complete program on its own: a script holding `}})` would close the
 * wrapper early and run the rest at global scope, with the real window and without strict mode.
 */
export function wrapFloorScript(code: string): string {
  try {
    parseJs(code, { ecmaVersion: "latest", sourceType: "script" });
  } catch {
    throw new Error("a floor script must be a complete program");
  }
  return `(function(${PARAMS}){"use strict";{\n${code}\n}}).call(undefined,${ARGS});`;
}

function collectScripts(node: TreeNode, out: ElementNode[]): void {
  if (defaultTreeAdapter.isElementNode(node) && node.tagName === "script") out.push(node);
  if ("childNodes" in node) for (const child of node.childNodes) collectScripts(child, out);
}

function attr(element: ElementNode, name: string): string | undefined {
  return element.attrs.find((item) => item.name === name)?.value;
}

function scriptText(element: ElementNode): string {
  return element.childNodes.map((child) => (defaultTreeAdapter.isTextNode(child) ? child.value : "")).join("");
}

function setScriptText(element: ElementNode, text: string): void {
  element.childNodes = [];
  defaultTreeAdapter.insertText(element, text);
}

function newScript(text: string, attrs: { name: string; value: string }[] = []): ElementNode {
  const script = defaultTreeAdapter.createElement("script", htmlNs.NS.HTML, attrs);
  setScriptText(script, text);
  return script;
}

export type AssembledFloor = {
  html: string;
  /** Space-separated CSP sources for the guard and every wrapped script. */
  scriptHashes: string;
  /** keccak256 of the assembled HTML. */
  hash: `0x${string}`;
};

/** Deterministic: the same raw floor and snapshot always give the same bytes. */
export function assembleFloor(rawHtml: string, snapshotJson: string | null): AssembledFloor {
  const doc = parse(rawHtml.replace(/\r\n/g, "\n"));
  const htmlElement = doc.childNodes.find((node): node is ElementNode => defaultTreeAdapter.isElementNode(node) && node.tagName === "html");
  if (!htmlElement) throw new Error("assembleFloor: no html element");
  const head = htmlElement.childNodes.find((node): node is ElementNode => defaultTreeAdapter.isElementNode(node) && node.tagName === "head");
  if (!head) throw new Error("assembleFloor: no head element");

  const hashes: string[] = [cspScriptHash(WALLET_GUARD)];
  const scripts: ElementNode[] = [];
  collectScripts(htmlElement, scripts);

  for (const script of scripts) {
    if (attr(script, "type") === "application/json") {
      if (attr(script, "id") === "snapshot") defaultTreeAdapter.detachNode(script);
      continue;
    }
    const wrapped = wrapFloorScript(scriptText(script));
    setScriptText(script, wrapped);
    hashes.push(cspScriptHash(wrapped));
  }

  const safeSnapshot = (snapshotJson ?? "{}").replace(/</g, "\\u003c");
  const first = head.childNodes[0] ?? null;
  const guardScript = newScript(WALLET_GUARD);
  const snapshotScript = newScript(safeSnapshot, [
    { name: "type", value: "application/json" },
    { name: "id", value: "snapshot" },
  ]);
  if (first) {
    defaultTreeAdapter.insertBefore(head, guardScript, first);
    defaultTreeAdapter.insertBefore(head, snapshotScript, first);
  } else {
    defaultTreeAdapter.appendChild(head, guardScript);
    defaultTreeAdapter.appendChild(head, snapshotScript);
  }

  const output = serialize(doc);
  return { html: output, scriptHashes: hashes.join(" "), hash: keccakText(output) };
}

/** The iframe attribute every floor is embedded with. No allow-same-origin, ever. */
export const FLOOR_FRAME_SANDBOX = "allow-scripts";

export function floorCsp(scriptHashes: string, frameAncestor: string): string {
  return [
    "sandbox allow-scripts",
    "default-src 'none'",
    `script-src ${scriptHashes}`,
    "style-src 'unsafe-inline'",
    "img-src data:",
    "font-src data:",
    "media-src data:",
    "connect-src 'none'",
    "frame-src 'none'",
    "worker-src 'none'",
    "object-src 'none'",
    "manifest-src 'none'",
    "form-action 'none'",
    "base-uri 'none'",
    `frame-ancestors ${frameAncestor}`,
  ].join("; ");
}

export function floorHeaders(scriptHashes: string, frameAncestor: string): Record<string, string> {
  return {
    "Content-Type": "text/html; charset=utf-8",
    "Content-Security-Policy": floorCsp(scriptHashes, frameAncestor),
    "X-Content-Type-Options": "nosniff",
    "Referrer-Policy": "no-referrer",
    "Permissions-Policy": "camera=(), microphone=(), geolocation=(), payment=(), usb=(), clipboard-read=()",
    "Cross-Origin-Resource-Policy": "cross-origin",
    "Cache-Control": "public, max-age=31536000, immutable",
  };
}

import { parse as parseJs } from "acorn";
import { ancestor as walkAncestor } from "acorn-walk";
import { defaultTreeAdapter, parse as parseHtml, type DefaultTreeAdapterTypes } from "parse5";
import { MAX_FLOOR_BYTES } from "@/lib/config";

export type Finding = { rule: string; detail: string };
export type JsNode = { type: string; start: number; end: number; [key: string]: unknown };
type TreeNode = DefaultTreeAdapterTypes.Node;

const ALLOWED_ELEMENTS = new Set("html head title meta style script body main section header footer nav article aside h1 h2 h3 h4 h5 h6 p div span ul ol li dl dt dd table thead tbody tfoot tr th td caption colgroup col form fieldset legend label input textarea select option optgroup button output progress meter details summary small strong em b i u s sub sup code pre kbd samp var mark abbr time blockquote q cite hr br img figure figcaption canvas a svg g path circle rect line polyline polygon ellipse text tspan defs lineargradient radialgradient stop clippath mask desc".split(" "));
const URL_ATTRS = new Set(["src", "href", "xlink:href", "action", "formaction", "srcset", "poster", "background", "ping", "data", "manifest", "cite", "longdesc"]);
const CORE_BANNED_NAMES = new Set(["location", "navigation", "history", "ownerDocument", "getRootNode", "write", "writeln"]);
const BANNED_CREATE = new Set(["iframe", "script", "object", "embed", "link", "base", "meta", "form", "frame", "frameset", "portal"]);

function isSafeUrl(value: string): boolean {
  const v = value.trim();
  return v === "" || v.startsWith("#") || /^data:(image|font)\//i.test(v);
}

function checkCss(css: string, where: string, findings: Finding[]): void {
  for (const match of css.matchAll(/url\(\s*(['"]?)([^'")]*)\1\s*\)/gi)) {
    const target = match[2].trim();
    if (!/^data:(image|font)\//i.test(target) && !target.startsWith("#")) {
      findings.push({ rule: "css-external", detail: `${where} loads ${target.slice(0, 80)}` });
    }
  }
  if (/@import/i.test(css)) findings.push({ rule: "css-import", detail: `${where} uses @import` });
}

export type ParsedFloor = { scripts: string[]; findings: Finding[] };

/** Walks the HTML, applies the structure rules and returns the inline scripts. */
export function parseFloor(html: string): ParsedFloor {
  const findings: Finding[] = [];
  if (Buffer.byteLength(html, "utf8") > MAX_FLOOR_BYTES) {
    findings.push({ rule: "file-size", detail: `the floor is over ${MAX_FLOOR_BYTES} bytes` });
  }
  const scripts: string[] = [];
  const visit = (node: TreeNode): void => {
    if (defaultTreeAdapter.isElementNode(node)) {
      const tag = node.tagName.toLowerCase();
      if (!ALLOWED_ELEMENTS.has(tag)) findings.push({ rule: "element", detail: `<${tag}> is not allowed` });
      for (const attribute of node.attrs) {
        const name = attribute.name.toLowerCase();
        if (name.startsWith("on")) findings.push({ rule: "inline-handler", detail: `<${tag} ${name}> is not allowed; use addEventListener` });
        if (name === "srcdoc") findings.push({ rule: "srcdoc", detail: `<${tag} srcdoc> is not allowed` });
        if (URL_ATTRS.has(name) && !isSafeUrl(attribute.value)) findings.push({ rule: "external-url", detail: `<${tag} ${name}="${attribute.value.slice(0, 80)}"> must be a #fragment or data: URL` });
        if (name === "style") checkCss(attribute.value, `<${tag} style>`, findings);
      }
      if (tag === "meta" && node.attrs.some((item) => item.name === "http-equiv")) findings.push({ rule: "meta-http-equiv", detail: "<meta http-equiv> is not allowed" });
      if (tag === "form" && node.attrs.some((item) => item.name === "action")) findings.push({ rule: "form-action", detail: "<form action> is not allowed" });
      const text = node.childNodes.map((child) => (defaultTreeAdapter.isTextNode(child) ? child.value : "")).join("");
      if (tag === "style") checkCss(text, "<style>", findings);
      if (tag === "script") {
        const type = node.attrs.find((item) => item.name === "type")?.value;
        if (node.attrs.some((item) => item.name === "src")) findings.push({ rule: "script-src", detail: "<script src> is not allowed; keep all code inline" });
        if (type === "application/json") {
          if (node.attrs.find((item) => item.name === "id")?.value !== "snapshot") findings.push({ rule: "script-type", detail: 'the only JSON script allowed has id="snapshot"' });
        } else if (type !== undefined && type !== "" && type !== "text/javascript") {
          findings.push({ rule: "script-type", detail: `<script type="${type}"> is not allowed` });
        } else {
          scripts.push(text);
        }
      }
    }
    if ("childNodes" in node) for (const child of node.childNodes) visit(child);
  };
  visit(parseHtml(html));
  if (scripts.length > 1) findings.push({ rule: "one-script", detail: `the floor has ${scripts.length} scripts; keep all code in one` });
  return { scripts, findings };
}

export function parseScript(code: string, findings: Finding[]): JsNode | null {
  try {
    return parseJs(code, { ecmaVersion: "latest", sourceType: "script" }) as unknown as JsNode;
  } catch (error) {
    findings.push({ rule: "js-syntax", detail: error instanceof Error ? error.message : "the script does not parse" });
    return null;
  }
}

/** The name of a member access, for both a.b and a["b"]. Null when it cannot be known statically. */
export function memberName(node: JsNode): string | null {
  const property = node.property as JsNode;
  if (!node.computed && property.type === "Identifier") return property.name as string;
  if (property.type === "Literal" && typeof property.value === "string") return property.value;
  return null;
}

export function checkCoreScript(ast: JsNode): Finding[] {
  const findings: Finding[] = [];
  walkAncestor(ast as never, {
    ImportExpression() {
      findings.push({ rule: "dynamic-import", detail: "import() is not allowed" });
    },
    Identifier(node: unknown, _state: unknown, ancestors: unknown) {
      const id = node as JsNode;
      const chain = ancestors as JsNode[];
      const parent = chain[chain.length - 2];
      const name = id.name as string;
      if (CORE_BANNED_NAMES.has(name)) findings.push({ rule: "banned-name", detail: `${name} is not allowed: a floor must not navigate` });
      if (name === "document") {
        const isStaticObject = parent?.type === "MemberExpression" && parent.object === id && !parent.computed;
        if (!isStaticObject) findings.push({ rule: "document-alias", detail: "document may only be used as document.something, never stored or indexed" });
      }
    },
    MemberExpression(node: unknown) {
      const member = node as JsNode;
      const name = memberName(member);
      if (name && CORE_BANNED_NAMES.has(name)) findings.push({ rule: "banned-name", detail: `.${name} is not allowed: a floor must not navigate` });
    },
    CallExpression(node: unknown) {
      const call = node as JsNode;
      const callee = call.callee as JsNode;
      if (callee.type !== "MemberExpression") return;
      const name = memberName(callee);
      if (name !== "createElement" && name !== "createElementNS") return;
      const args = call.arguments as JsNode[];
      const tagArg = name === "createElement" ? args[0] : args[1];
      if (!tagArg || tagArg.type !== "Literal" || typeof tagArg.value !== "string") {
        findings.push({ rule: "create-element", detail: `${name} needs a literal tag name` });
      } else if (BANNED_CREATE.has(tagArg.value.toLowerCase())) {
        findings.push({ rule: "create-element", detail: `${name}("${tagArg.value}") is not allowed` });
      }
    },
  } as never);
  return findings;
}

import { checkCoreScript, parseFloor, parseScript, type Finding, type JsNode } from "@/lib/floor-gate/static-core";

export type { Finding } from "@/lib/floor-gate/static-core";

type ScriptRule = (ast: JsNode) => Finding[];

const SCRIPT_RULES: ScriptRule[] = [checkCoreScript];

/** Runs every static rule on a raw floor. An empty list means it passed. */
export function runStaticGate(html: string): Finding[] {
  const { scripts, findings } = parseFloor(html);
  for (const code of scripts) {
    const ast = parseScript(code, findings);
    if (!ast) continue;
    for (const rule of SCRIPT_RULES) findings.push(...rule(ast));
  }
  return findings;
}

/** The core rules alone, for tests that prove which rule set catches what. */
export function runCoreGate(html: string): Finding[] {
  const { scripts, findings } = parseFloor(html);
  for (const code of scripts) {
    const ast = parseScript(code, findings);
    if (ast) findings.push(...checkCoreScript(ast));
  }
  return findings;
}

import { createServer } from "node:http";
import type { AddressInfo } from "node:net";
import type { Browser, Frame, Page } from "playwright-core";
import { FLOOR_FRAME_SANDBOX, floorHeaders } from "@/lib/floor-gate/headers";
import type { FloorSpec, FloorStep, FloorTest } from "@/lib/floor-gate/floor-spec";

export type ProviderMode = "configurable" | "non-configurable";
export type Violation = { kind: "provider-touched" | "network" | "navigation" | "dialog" | "new-window"; detail: string };
export type TestResult = { name: string; pass: boolean; detail: string | null };
export type BrowserGateResult = {
  violations: Violation[];
  errors: string[];
  tests: TestResult[];
  screenshots: { wide: string; narrow: string } | null;
};

const PROBE = "__NC_PROBE__";
const STEP_TIMEOUT_MS = 3000;

/** A stand-in for a wallet extension: installed before any page script, in every frame. */
export function fakeProviderScript(mode: ProviderMode): string {
  const install = mode === "configurable"
    ? "window.ethereum = provider;"
    : 'Object.defineProperty(window, "ethereum", { value: provider, configurable: false, writable: false, enumerable: true });';
  return `(function () {
    var touched = function (what) { console.log("${PROBE} " + what); };
    var provider = new Proxy({ isMetaMask: true, request: function () { touched("request"); return Promise.resolve([]); } }, {
      get: function (target, key) { touched("get:" + String(key)); return target[key]; }
    });
    ${install}
    // Held from before the guard runs, as a real extension could, so only the guard's listener stops it.
    var announce = window.dispatchEvent.bind(window);
    window.addEventListener("eip6963:requestProvider", function () {
      announce(new CustomEvent("eip6963:announceProvider", { detail: { info: { rdns: "test.wallet" }, provider: provider } }));
    });
  })();`;
}

function parentPage(floorPath: string): string {
  return `<!doctype html><html><head><meta charset="utf-8"><title>floor gate</title><style>html,body{margin:0;height:100%}iframe{display:block;width:100%;height:100vh;border:0}</style></head><body><iframe id="floor" title="floor" sandbox="${FLOOR_FRAME_SANDBOX}" allow="" referrerpolicy="no-referrer" src="${floorPath}"></iframe></body></html>`;
}

async function runStep(frame: Frame, step: FloorStep): Promise<void> {
  const locator = frame.locator(step.selector).first();
  switch (step.op) {
    case "fill":
      await locator.fill(step.value, { timeout: STEP_TIMEOUT_MS });
      return;
    case "click":
      await locator.click({ timeout: STEP_TIMEOUT_MS });
      return;
    case "select":
      await locator.selectOption(step.value, { timeout: STEP_TIMEOUT_MS });
      return;
    case "press":
      await locator.press(step.key, { timeout: STEP_TIMEOUT_MS });
      return;
    case "expectText": {
      const text = (await locator.textContent({ timeout: STEP_TIMEOUT_MS })) ?? "";
      if (!text.includes(step.contains)) throw new Error(`expected "${step.selector}" to contain "${step.contains}", found "${text.slice(0, 120)}"`);
      return;
    }
    case "expectValue": {
      const value = await locator.inputValue({ timeout: STEP_TIMEOUT_MS });
      if (value !== step.equals) throw new Error(`expected "${step.selector}" to equal "${step.equals}", found "${value.slice(0, 120)}"`);
      return;
    }
    case "expectVisible":
      await locator.waitFor({ state: "visible", timeout: STEP_TIMEOUT_MS });
      return;
    case "expectCount": {
      const count = await frame.locator(step.selector).count();
      if (count !== step.count) throw new Error(`expected ${step.count} of "${step.selector}", found ${count}`);
      return;
    }
  }
}

async function loadFloor(page: Page, origin: string, settleMs: number): Promise<Frame> {
  await page.goto(`${origin}/parent`, { waitUntil: "load" });
  await page.waitForTimeout(settleMs);
  const frame = page.frames().find((candidate) => candidate.url().endsWith("/f/floor"));
  if (!frame) throw new Error("the floor frame did not load");
  return frame;
}

async function runTest(page: Page, origin: string, settleMs: number, test: FloorTest): Promise<TestResult> {
  try {
    const frame = await loadFloor(page, origin, settleMs);
    for (const step of test.steps) await runStep(frame, step);
    return { name: test.name, pass: true, detail: null };
  } catch (error) {
    return { name: test.name, pass: false, detail: error instanceof Error ? error.message.slice(0, 300) : "test failed" };
  }
}

export type BrowserGateOptions = {
  html: string;
  scriptHashes: string;
  spec: Pick<FloorSpec, "tests">;
  launch: () => Promise<Browser>;
  providerMode?: ProviderMode;
  settleMs?: number;
  screenshots?: boolean;
};

/** Runs an assembled floor under production headers with a fake wallet. See the spec, section 6. */
export async function runBrowserGate(options: BrowserGateOptions): Promise<BrowserGateResult> {
  const { html, scriptHashes, spec, launch, providerMode = "non-configurable", settleMs = 600, screenshots = false } = options;
  const server = createServer((request, response) => {
    const path = new URL(request.url ?? "/", "http://x").pathname;
    if (path === "/parent") {
      response.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
      response.end(parentPage("/f/floor"));
    } else if (path === "/f/floor") {
      const { address, port } = server.address() as AddressInfo;
      response.writeHead(200, floorHeaders(scriptHashes, `http://${address}:${port}`));
      response.end(html);
    } else {
      response.writeHead(404);
      response.end("not found");
    }
  });

  } finally {
    await browser.close();
    server.close();

import { ancestor as walkAncestor, simple as walkSimple } from "acorn-walk";
import { memberName, type Finding, type JsNode } from "@/lib/floor-gate/static-core";

const BANNED_ANYWHERE = new Set(["fetch", "XMLHttpRequest", "WebSocket", "EventSource", "RTCPeerConnection", "sendBeacon", "importScripts", "Worker", "SharedWorker", "serviceWorker", "postMessage", "MessageChannel", "BroadcastChannel", "localStorage", "sessionStorage", "indexedDB", "cookie", "caches", "eval", "Function", "__proto__", "globalThis", "defaultView", "contentWindow", "contentDocument", "frameElement", "srcElement", "ethereum", "web3", "solana", "phantom", "execCommand", "__ncw"]);
const BANNED_BARE_GLOBALS = new Set(["self", "top", "parent", "frames", "open", "opener", "chrome", "external"]);
const WINDOW_ALLOWLIST = new Set(["addEventListener", "removeEventListener", "innerWidth", "innerHeight", "devicePixelRatio", "matchMedia", "getComputedStyle", "requestAnimationFrame", "cancelAnimationFrame", "setTimeout", "clearTimeout", "setInterval", "clearInterval"]);
const BANNED_STRINGS = ["ethereum", "web3", "solana", "phantom", "eip6963", "metamask", "http://", "https://", "javascript:"];

function collectDeclaredNames(ast: JsNode): Set<string> {
  const declared = new Set<string>();
  const addPattern = (pattern: JsNode | null): void => {
    if (!pattern) return;
    if (pattern.type === "Identifier") declared.add(pattern.name as string);
    if (pattern.type === "ObjectPattern") for (const property of pattern.properties as JsNode[]) addPattern((property.value ?? property.argument) as JsNode);
    if (pattern.type === "ArrayPattern") for (const element of pattern.elements as (JsNode | null)[]) addPattern(element);
    if (pattern.type === "AssignmentPattern") addPattern(pattern.left as JsNode);
    if (pattern.type === "RestElement") addPattern(pattern.argument as JsNode);
  };
  const addFunction = (fn: JsNode): void => {
    if (fn.id) addPattern(fn.id as JsNode);
    for (const param of fn.params as JsNode[]) addPattern(param);
  };
  walkSimple(ast as never, {
    VariableDeclarator(node: unknown) {
      addPattern((node as JsNode).id as JsNode);
    },
    FunctionDeclaration(node: unknown) {
      addFunction(node as JsNode);
    },
    FunctionExpression(node: unknown) {
      addFunction(node as JsNode);
    },
    ArrowFunctionExpression(node: unknown) {
      addFunction(node as JsNode);
    },
    ClassDeclaration(node: unknown) {
      addPattern((node as JsNode).id as JsNode);
    },
    CatchClause(node: unknown) {
      addPattern((node as JsNode).param as JsNode);
    },
  } as never);
  return declared;
}

/** Rules the runtime already contains. They exist to give the agent clearer, earlier messages. */
export function checkExtendedScript(ast: JsNode): Finding[] {
  const findings: Finding[] = [];
  const declared = collectDeclaredNames(ast);
  const checkString = (value: string): void => {
    for (const banned of BANNED_STRINGS) {
      if (value.includes(banned)) findings.push({ rule: "banned-string", detail: `a string contains "${banned}"` });
    }
  };
  walkAncestor(ast as never, {
    Identifier(node: unknown, _state: unknown, ancestors: unknown) {
      const id = node as JsNode;
      const chain = ancestors as JsNode[];
      const parent = chain[chain.length - 2];
      const name = id.name as string;
      if (BANNED_ANYWHERE.has(name)) findings.push({ rule: "banned-name", detail: `${name} is not available in a floor` });
      if (BANNED_BARE_GLOBALS.has(name) && !declared.has(name)) findings.push({ rule: "banned-global", detail: `${name} is not available in a floor` });
      if (name === "window") {
        const isStaticObject = parent?.type === "MemberExpression" && parent.object === id && !parent.computed;
        const member = isStaticObject ? memberName(parent) : null;
        if (!member || !WINDOW_ALLOWLIST.has(member)) findings.push({ rule: "window-member", detail: `window${member ? `.${member}` : ""} is not available in a floor` });
      }
    },
    MemberExpression(node: unknown) {
      const member = node as JsNode;
      const name = memberName(member);
      if (name && BANNED_ANYWHERE.has(name)) findings.push({ rule: "banned-name", detail: `.${name} is not available in a floor` });
      if (name === "constructor") findings.push({ rule: "banned-name", detail: ".constructor is not available in a floor" });
    },
    Literal(node: unknown) {
      const literal = node as JsNode;
      if (typeof literal.value === "string") checkString(literal.value);
    },
    TemplateElement(node: unknown) {
      const element = node as JsNode;
      const cooked = (element.value as { cooked?: string | null }).cooked;
      if (cooked) checkString(cooked);
    },
  } as never);
  return findings;
}

import { checkExtendedScript } from "@/lib/floor-gate/static-extended";

export type { Finding } from "@/lib/floor-gate/static-core";

type ScriptRule = (ast: JsNode) => Finding[];

const SCRIPT_RULES: ScriptRule[] = [checkCoreScript, checkExtendedScript];

/** Runs every static rule on a raw floor. An empty list means it passed. */
export function runStaticGate(html: string): Finding[] {
  const { scripts, findings } = parseFloor(html);
  for (const code of scripts) {
    const ast = parseScript(code, findings);
    if (!ast) continue;
    for (const rule of SCRIPT_RULES) findings.push(...rule(ast));
  }
  return findings;
}

/** The core rules alone, for tests that prove which rule set catches what. */
export function runCoreGate(html: string): Finding[] {
  const { scripts, findings } = parseFloor(html);
  for (const code of scripts) {
    const ast = parseScript(code, findings);
    if (ast) findings.push(...checkCoreScript(ast));
  }
  return findings;
}

# Floor rules

A floor is one small working tool, in one HTML file. It runs in a locked sandbox in a visitor's browser. Build inside these rules, or the shift fails.

## Deliverables

Write both files with `write_file`:

1. `index.html`: the whole tool. Start from `template.html`.
2. `floor.json`: its name, one line, how to use it, and its tests (see below).

Before you finish, call `run_floor_check` and fix everything it reports. It prints `CHECK PASS` when the files pass the static rules. You are done only when it does.

That check reads your files and does not run them. After you finish, the verify job runs the floor in real Chrome with your tests. An uncaught error or a failing test fails the shift.

## What a floor can do

- Use HTML, CSS and one inline `<script>`.
- Use the DOM, canvas, inline SVG, `data:` images and `innerHTML`.
- Keep state in memory while the page is open.
- Read the data snapshot: `JSON.parse(document.getElementById("snapshot").textContent)`. It holds public figures fetched for this shift. The pipeline inserts it, so a floor built without it still runs. Treat every field as possibly missing.

## What a floor cannot do

- Use the network. There is no `fetch`, no external script, style, image or font.
- Store anything. There are no cookies, `localStorage` or `indexedDB`.
- Touch a wallet.
- Navigate. There are no links out, no `location`, no `history`, no `form action`.
- Open windows or dialogs. There is no `alert`, `confirm`, `prompt` or `window.open`.
- Use `eval`, `new Function`, or string timers.
- Use more than one `<script>`, inline `on...` attributes, or `<iframe>`, `<object>`, `<embed>`, `<link>` or `<base>`.
- Assign handlers such as `onload` or `onerror` on `window` or `document.body`. They do nothing here, so use `addEventListener`.
- Exceed 256 KB.

Use only `document.something`, never a stored or computed `document`. Use only these on `window`: `addEventListener`, `removeEventListener`, `innerWidth`, `innerHeight`, `devicePixelRatio`, `matchMedia`, `getComputedStyle`, `requestAnimationFrame`, `cancelAnimationFrame`, `setTimeout`, `clearTimeout`, `setInterval`, `clearInterval`.

Plain words such as "Ethereum" are fine in labels. The lowercase names `ethereum`, `web3`, `solana` and `phantom` are not, in code or in strings, and neither are `http://` or `https://`.

import { getDb } from "@/lib/db";
import { floorHeaders } from "@/lib/floor-gate/headers";

type FloorRow = { html: string; script_hashes: string; status: string };

/** Who may frame this floor: the main site, or on the main host, whoever asked at this host. */
function frameAncestor(request: Request): string {
  const configured = process.env.NEXT_PUBLIC_SITE_ORIGIN;
  if (configured) return configured;
  const host = request.headers.get("x-forwarded-host") ?? request.headers.get("host");
  const url = new URL(request.url);
  const protocol = request.headers.get("x-forwarded-proto") ?? url.protocol.replace(":", "");
  return host ? `${protocol}://${host}` : url.origin;
}

export async function GET(request: Request, context: { params: Promise<{ hash: string }> }): Promise<Response> {
  const { hash } = await context.params;
  const db = await getDb();
  const result = await db.query<FloorRow>("select html, script_hashes, status from floors where hash = $1", [hash]);
  const floor = result.rows[0];
  if (!floor) return new Response("Not found", { status: 404 });
  if (floor.status === "vetoed") return new Response("This floor was vetoed. Its receipt stays public.", { status: 410 });
  return new Response(floor.html, { status: 200, headers: floorHeaders(floor.script_hashes, frameAncestor(request)) });
}

import { NextResponse, type NextRequest } from "next/server";

/** With a floors hostname set, floors live only there and everything else lives only on the main host. */
export function routeByHost(pathname: string, host: string | null, floorsOrigin: string | undefined): "next" | "not-found" {
  if (!floorsOrigin) return "next";
  const floorsHost = new URL(floorsOrigin).host;
  const onFloorsHost = host === floorsHost;
  const isFloorPath = pathname.startsWith("/f/");
  if (onFloorsHost && !isFloorPath) return "not-found";
  if (!onFloorsHost && isFloorPath) return "not-found";
  return "next";
}

export function proxy(request: NextRequest): NextResponse {
  const route = routeByHost(request.nextUrl.pathname, request.headers.get("host"), process.env.FLOORS_ORIGIN);
  if (route === "not-found") return new NextResponse("Not found", { status: 404 });
  return NextResponse.next();
}

export const config = {
  matcher: ["/((?!_next/|taste/|favicon.ico).*)"],
};
